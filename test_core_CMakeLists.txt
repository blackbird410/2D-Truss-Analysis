cmake_minimum_required(VERSION 3.20)
project(TrussAnalysisCore VERSION 2.0.0 LANGUAGES CXX)

# Set C++ standard
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# Set build type
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release)
endif()

# Compiler-specific options
if(MSVC)
    add_compile_options(/W4)
else()
    add_compile_options(-Wall -Wextra -Wpedantic)
endif()

# Find required packages (only Eigen3 for now)
find_package(Eigen3 REQUIRED)

# Create library for core functionality
add_library(TrussCore STATIC
    src/core/Logger.cpp
    src/core/Logger.hpp
    src/core/Types.hpp
    src/core/Node.cpp
    src/core/Node.hpp
    src/core/Member.cpp
    src/core/Member.hpp
    src/core/Truss.cpp
    src/core/Truss.hpp
    src/core/AnalysisEngine.cpp
    src/core/AnalysisEngine.hpp
)

target_include_directories(TrussCore PUBLIC
    src/core
)

target_link_libraries(TrussCore PUBLIC
    Eigen3::Eigen
)

# Set target properties
set_target_properties(TrussCore PROPERTIES
    CXX_STANDARD 20
    CXX_STANDARD_REQUIRED ON
)

# Test executable
add_executable(test_analysis
    test_analysis.cpp
)

target_link_libraries(test_analysis PRIVATE
    TrussCore
)
