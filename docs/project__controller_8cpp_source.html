35void ProjectController::onNewProjectRequested() {
36 if (m_isDirty && m_trussHandle != 0) {
37 const bool confirmed = m_confirm.confirm(
38 QStringLiteral("Unsaved Changes"),
39 QStringLiteral("The current project has unsaved changes.\n"
40 "Discard them and create a new project?"));
41 if (!confirmed)
42 return;
43 }
44
45 if (m_trussHandle != 0) {
46 m_facade.clearTruss(m_trussHandle);
47 }
48
49 auto result = m_facade.createTruss("Untitled Truss");
50 if (result) {
51 m_trussHandle = result.value;
52 m_currentFilePath.clear();
53 m_isDirty = false;
54 emit trussCreated(m_trussHandle);
55 } else {
56 emit operationFailed(
57 QString::fromStdString("Failed to create truss: " + result.errorMessage));
58 }
59}

65void ProjectController::onOpenFileRequested() {
66 if (m_isDirty && m_trussHandle != 0) {
67 const bool confirmed = m_confirm.confirm(
68 QStringLiteral("Unsaved Changes"),
69 QStringLiteral("The current project has unsaved changes.\n"
70 "Discard them and open a different file?"));
71 if (!confirmed)
72 return;
73 }
74
75 const QString filePath = QFileDialog::getOpenFileName(
76 nullptr,
77 QStringLiteral("Open Truss Project"),
78 QStringLiteral("."),
79 QStringLiteral("Truss files (*.json *.xml);;All files (*)"));
80
81 if (filePath.isEmpty())
82 return; // user cancelled
83
84 auto result = m_facade.loadTruss(std::filesystem::path{filePath.toStdString()});
85 if (result) {
86 if (m_trussHandle != 0) {
87 m_facade.clearTruss(m_trussHandle);
88 }
89 m_trussHandle = result.value;
90 m_currentFilePath = filePath;
91 m_isDirty = false;
92 emit trussLoaded(m_trussHandle, filePath);
93 } else {
94 emit operationFailed(QString::fromStdString("Failed to load file: " + result.errorMessage));
95 }
96}
