44void ConsolePresenter::displayAnalysisResults(const IAnalysisResultsView& resultsView) const {
45 m_output.info("");
46 m_output.info("Analysis Results:");
47 m_output.info(" Converged: " + std::string(resultsView.hasConverged() ? "Yes" : "No"));
48
49 {
50 std::ostringstream oss;
51 oss << " Maximum displacement: " << std::fixed << std::setprecision(4)
52 << resultsView.getMaxDisplacement() * 1000 << " mm";
53 m_output.info(oss.str());
54 }
55
56 {
57 std::ostringstream oss;
58 oss << " Maximum stress: " << std::fixed << std::setprecision(2)
59 << resultsView.getMaxStress() / 1e6 << " MPa";
60 m_output.info(oss.str());
61 }
62
63 {
64 std::ostringstream oss;
65 oss << " Total strain: " << std::scientific << std::setprecision(3)
66 << resultsView.getTotalStrain();
67 m_output.info(oss.str());
68 }
69
70 {
71 std::ostringstream oss;
72 oss << " Matrix condition number: " << std::fixed << std::setprecision(1)
73 << resultsView.getConditionNumber();
74 m_output.info(oss.str());
75 }
76
77 // Display member forces
78 const auto& memberForces = resultsView.getMemberForces();
79 const auto& memberStresses = resultsView.getMemberStresses();
80 const auto& utilizationRatios = resultsView.getUtilizationRatios();
81
82 m_output.info("");
83 m_output.info("Member Forces:");
84 for (size_t i = 0; i < memberForces.size(); ++i) {
85 Real force = memberForces[i];
86 Real stress = memberStresses[i];
87 Real utilization = utilizationRatios[i];
88
89 std::ostringstream oss;
90 oss << " Member " << (i + 1) << ": Force = " << std::fixed << std::setprecision(2)
91 << force / 1000 << " kN" << ", Stress = " << std::fixed << std::setprecision(1)
92 << stress / 1e6 << " MPa" << ", Utilization = " << std::fixed << std::setprecision(1)
93 << utilization * 100 << "%" << (force > 0 ? " (Tension)" : " (Compression)");
94 m_output.info(oss.str());
95 }
96
97 // Display node displacements
98 const auto& displacements = resultsView.getDisplacements();
99 m_output.info("");
100 m_output.info("Node Displacements:");
101
102 // Note: This is a simplified display. Full implementation would iterate through nodes.
103 // For now, display first few DOFs as examples.
104 size_t numDofs = std::min(displacements.size(), size_t(6)); // Show first 3 nodes (6 DOFs)
105 for (size_t i = 0; i < numDofs; i += 2) {
106 Real dispX = displacements[i] * 1000; // mm
107 Real dispY = displacements[i + 1] * 1000; // mm
108
109 std::ostringstream oss;
110 oss << " Node " << (i / 2 + 1) << ": (" << std::fixed << std::setprecision(4) << dispX
111 << ", " << dispY << ") mm";
112 m_output.info(oss.str());
113 }
114}
