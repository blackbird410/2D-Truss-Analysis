20std::shared_ptr<Truss> TrussAssembler::assembleTruss(const interfaces::TrussDTO& dto) {
21 auto truss = std::make_shared<Truss>(dto.name);
22
23 // Map to track created nodes by their DTO IDs
24 std::unordered_map<NodeId, std::shared_ptr<Node>> nodeMap;
25
26 // Phase 1: Create all nodes
27 for (const auto& nodeDTO : dto.nodes) {
28 auto node = std::make_shared<Node>(
29 nodeDTO.id, Point2D{nodeDTO.x, nodeDTO.y}, nodeDTO.support);
30
31 // Apply forces if present
32 if (nodeDTO.fx != 0.0 || nodeDTO.fy != 0.0) {
33 node->setAppliedForce(Force2D(nodeDTO.fx, nodeDTO.fy));
34 }
35
36 truss->addNode(node);
37 nodeMap[nodeDTO.id] = node;
38 }
39
40 // Phase 2: Create all members
41 for (const auto& memberDTO : dto.members) {
42 // Validate that referenced nodes exist
43 auto startIt = nodeMap.find(memberDTO.startNodeId);
44 auto endIt = nodeMap.find(memberDTO.endNodeId);
45
46 if (startIt == nodeMap.end()) {
47 std::ostringstream oss;
48 oss << "Member " << memberDTO.id << " references non-existent start node "
49 << memberDTO.startNodeId;
50 throw std::invalid_argument(oss.str());
51 }
52
53 if (endIt == nodeMap.end()) {
54 std::ostringstream oss;
55 oss << "Member " << memberDTO.id << " references non-existent end node "
56 << memberDTO.endNodeId;
57 throw std::invalid_argument(oss.str());
58 }
59
60 // Create material and section properties
61 MaterialProperties material;
62 material.youngModulus = memberDTO.youngModulus;
63 material.yieldStrength = memberDTO.yieldStrength;
64 material.density = memberDTO.density;
65 material.name = "Material_" + std::to_string(memberDTO.id);
66
67 SectionProperties section;
68 section.area = memberDTO.area;
69 section.designation = "Section_" + std::to_string(memberDTO.id);
70
71 // Add member to truss
72 truss->addMember(startIt->second, endIt->second, material, section);
73 }
74
75 return truss;
76}

78interfaces::TrussDTO TrussAssembler::createDTO(const interfaces::ITrussView& truss) {
79 interfaces::TrussDTO dto(truss.getName());
80
81 // Convert NodeViews to NodeDTOs
82 const auto& nodeViews = truss.getNodeViews();
83 dto.nodes.reserve(nodeViews.size());
84
85 for (const auto& nodeView : nodeViews) {
86 dto.nodes.emplace_back(
87 nodeView.id, nodeView.x, nodeView.y, nodeView.support, nodeView.fx, nodeView.fy);
88 }
89
90 // Convert MemberViews to MemberDTOs
91 const auto& memberViews = truss.getMemberViews();
92 dto.members.reserve(memberViews.size());
93
94 for (const auto& memberView : memberViews) {
95 dto.members.emplace_back(memberView.id,
96 memberView.startNodeId,
97 memberView.endNodeId,
98 memberView.label,
99 memberView.youngModulus,
100 memberView.yieldStrength,
101 memberView.density,
102 memberView.area);
103 }
104
105 return dto;
106}
