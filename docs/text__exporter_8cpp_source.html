25bool TextExporter::exportResults(const ITrussView& truss,
26 const IAnalysisResultsView& results,
27 const std::filesystem::path& filePath,
28 const ExportOptions& options) {
29 std::ofstream file(filePath);
30 if (!file.is_open()) {
31 m_lastError = "Cannot open file for writing: " + filePath.string();
32 return false;
33 }
34
35 try {
36 // Document header
37 writeHeader(file, truss);
38
39 // ALL 8 sections MUST be represented (placeholder if not implemented)
40 if (options.includeGeometry) {
41 writeGeometrySection(file, truss, options);
42 }
43
44 if (options.includeProperties) {
45 writePropertiesSection(file, truss, options);
46 }
47
48 if (options.includeLoads) {
49 writeLoadsSection(file, truss, options);
50 }
51
52 if (options.includeDisplacements && results.getDisplacements().size() > 0) {
53 writeDisplacementsSection(file, results, options);
54 }
55
56 if (options.includeMemberForces && !results.getMemberForces().empty()) {
57 writeMemberForcesSection(file, results, options);
58 }
59
60 if (options.includeReactions && !results.getReactions().empty()) {
61 writeReactionsSection(file, results, options);
62 }
63
64 if (options.includeMetadata) {
65 writeMetadataSection(file, results, options);
66 }
67
68 // Document footer
69 writeSeparator(file);
70 file << "End of Report\n";
71
72 file.close();
73 return true;
74
75 } catch (const std::exception& e) {
76 m_lastError = "Export failed: " + std::string(e.what());
77 return false;
78 }
79}

111void TextExporter::writeGeometrySection(std::ostream& os,
112 const ITrussView& truss,
113 const ExportOptions& options) {
114 writeSectionHeader(os, "GEOMETRY");
115
116 // Nodes subsection
117 os << "\nNodes:\n";
118 os << " " << std::left << std::setw(8) << "Node ID" << std::setw(15) << "X (m)"
119 << std::setw(15) << "Y (m)" << std::setw(15) << "Support Type" << "\n";
120 os << " " << std::string(53, '-') << "\n";
121
122 for (const auto& node : truss.getNodeViews()) {
123 std::string supportType;
124 switch (node.support) {
125 case core::SupportType::Free:
126 supportType = "Free";
127 break;
128 case core::SupportType::Pinned:
129 supportType = "Pinned";
130 break;
131 case core::SupportType::RollerX:
132 supportType = "Roller-X";
133 break;
134 case core::SupportType::RollerY:
135 supportType = "Roller-Y";
136 break;
137 default:
138 supportType = "Unknown";
139 break;
140 }
141
142 os << " " << std::left << std::setw(8) << node.id << std::setw(15)
143 << truss::utils::string::formatReal(
144 node.x, options.precision, options.useScientificNotation)
145 << std::setw(15)
146 << truss::utils::string::formatReal(
147 node.y, options.precision, options.useScientificNotation)
148 << std::setw(15) << supportType << "\n";
149 }
150
151 // Members subsection
152 os << "\nMembers:\n";
153 os << " " << std::left << std::setw(10) << "Member ID" << std::setw(12) << "Start Node"
154 << std::setw(12) << "End Node" << std::setw(15) << "Length (m)" << "\n";
155 os << " " << std::string(49, '-') << "\n";
156
157 for (const auto& member : truss.getMemberViews()) {
158 os << " " << std::left << std::setw(10) << member.id << std::setw(12) << member.startNodeId
159 << std::setw(12) << member.endNodeId << std::setw(15)
160 << truss::utils::string::formatReal(
161 member.length, options.precision, options.useScientificNotation)
162 << "\n";
163 }
164}

195void TextExporter::writeLoadsSection(std::ostream& os,
196 const ITrussView& truss,
197 const ExportOptions& options) {
198 writeSectionHeader(os, "APPLIED LOADS");
199
200 os << "\n " << std::left << std::setw(10) << "Node ID" << std::setw(20) << "Fx (N)"
201 << std::setw(20) << "Fy (N)" << "\n";
202 os << " " << std::string(50, '-') << "\n";
203
204 bool hasLoads = false;
205 for (const auto& node : truss.getNodeViews()) {
206 // Only export nodes with non-zero forces
207 if (node.fx != 0.0 || node.fy != 0.0) {
208 os << " " << std::left << std::setw(10) << node.id << std::setw(20)
209 << truss::utils::string::formatReal(
210 node.fx, options.precision, options.useScientificNotation)
211 << std::setw(20)
212 << truss::utils::string::formatReal(
213 node.fy, options.precision, options.useScientificNotation)
214 << "\n";
215 hasLoads = true;
216 }
217 }
218
219 if (!hasLoads) {
220 os << " (No applied loads)\n";
221 }
