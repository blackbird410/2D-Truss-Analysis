46Result<TrussHandle> TrussApplicationService::loadTruss(const std::filesystem::path& filepath,
47 FileFormat format) {
48 try {
49 // Create appropriate reader
50 auto reader = FileIOFactory::createReader(format);
51
52 // Read DTO from file
53 FileIOOptions options;
54 options.validateOnRead = false; // We'll validate after assembly
55 auto dto = reader->read(filepath, options);
56
57 // Assemble Domain model from DTO
58 auto truss = assembly::TrussAssembler::assembleTruss(dto);
59
60 // Validate assembled truss
61 auto validationResult = validation::TrussValidator::validate(*truss);
62 if (!validationResult.isValid()) {
63 std::string errorMsg = "Truss loaded but failed validation: " +
64 validationResult.getSummary();
65
66 // Append detailed error messages
67 auto errorMessages = validationResult.getErrorMessages();
68 if (!errorMessages.empty()) {
69 errorMsg += "\nErrors:\n";
70 for (const auto& msg : errorMessages) {
71 errorMsg += " - " + msg + "\n";
72 }
73 }
74
75 return Result<TrussHandle>::Failure(errorMsg);
76 }
77
78 // Store and return handle
79 TrussHandle handle = generateHandle();
80 m_trusses[handle] = truss;
81 m_modifiedFlags[handle] = false; // Loaded truss, not modified yet
82 return Result<TrussHandle>::Success(handle);
83
84 } catch (const FileIOException& e) {
85 return Result<TrussHandle>::Failure(std::string("File I/O error: ") + e.what());
86 } catch (const std::exception& e) {
87 return Result<TrussHandle>::Failure(std::string("Failed to load truss: ") + e.what());
88 }
89}

107Result<bool> TrussApplicationService::saveTruss(TrussHandle handle,
108 const std::filesystem::path& filepath,
109 FileFormat format,
110 bool overwrite) {
111 try {
112 // Validate handle
113 if (!isValidTrussHandle(handle)) {
114 return Result<bool>::Failure("Invalid truss handle");
115 }
116
117 auto truss = m_trusses[handle];
118
119 // Validate truss before saving
120 auto validationResult = validation::TrussValidator::validate(*truss);
121 if (!validationResult.isValid()) {
122 return Result<bool>::Failure("Cannot save invalid truss: " +
123 validationResult.getSummary());
124 }
125
126 // Create DTO from Domain model
127 auto dto = assembly::TrussAssembler::createDTO(*truss);
128
129 // Create appropriate writer
130 auto writer = FileIOFactory::createWriter(format);
131
132 // Write to file
133 FileIOOptions options;
134 options.overwriteExisting = overwrite;
135 options.prettyPrint = true;
136 options.includeMetadata = true;
137
138 writer->write(dto, filepath, options);
139 markAsSaved(handle); // Mark as saved after successful write
140 return Result<bool>::Success(true);
141
142 } catch (const FileIOException& e) {
143 return Result<bool>::Failure(std::string("File I/O error: ") + e.what());
144 } catch (const std::exception& e) {
145 return Result<bool>::Failure(std::string("Failed to save truss: ") + e.what());
146 }

217Result<MemberId> TrussApplicationService::addMember(TrussHandle handle,
218 NodeId startNodeId,
219 NodeId endNodeId,
220 const MaterialSpec& materialSpec,
221 const SectionSpec& sectionSpec) {
222 try {
223 if (!isValidTrussHandle(handle)) {
224 return Result<MemberId>::Failure("Invalid truss handle");
225 }
226
227 // Convert DTOs to Domain types with sensible defaults
228 MaterialProperties material{
229 materialSpec.youngsModulusPa, // E
230 7850.0, // density (default for steel)
231 materialSpec.youngsModulusPa * 0.00125, // yield strength (estimate)
232 materialSpec.youngsModulusPa * 0.002, // ultimate strength (estimate)
233 materialSpec.name // name
234 };
235
236 SectionProperties section{
237 sectionSpec.areaM2, // area
238 sectionSpec.areaM2 * sectionSpec.areaM2 / 12.0, // moment of inertia (estimate)
239 sectionSpec.areaM2, // shear area
240 sectionSpec.profile // designation
241 };
242
243 auto& truss = *m_trusses[handle];
244 MemberPtr member = truss.addMember(startNodeId, endNodeId, material, section);
245 markAsModified(handle);
246
247 return Result<MemberId>::Success(member->getId());
248
249 } catch (const std::exception& e) {
250 return Result<MemberId>::Failure(std::string("Failed to add member: ") + e.what());
251 }
