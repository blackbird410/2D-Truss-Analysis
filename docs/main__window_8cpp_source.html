76 // Install an application-level event filter to centralize dispatch of
77 // single-key shortcuts (N, M, Esc, Z, Delete) that cannot be registered
78 // via QAction::setShortcut() reliably — child widgets such as QTableView
79 // and QDoubleSpinBox can block Qt::WindowShortcut resolution for printable
80 // keys. The filter is removed in ~MainWindow() to prevent dangling-pointer

97void MainWindow::closeEvent(QCloseEvent* event) {
98 if (m_controller->state().isDirty) {
99 const auto reply = QMessageBox::question(
100 this,
101 QStringLiteral("Unsaved Changes"),
102 QStringLiteral("The project has unsaved changes.\n"
103 "Do you want to quit without saving?"),
104 QMessageBox::Discard | QMessageBox::Cancel,
105 QMessageBox::Cancel);
106 if (reply != QMessageBox::Discard) {
107 event->ignore();
108 return;
109 }
110 }
111 event->accept();
112}

624void MainWindow::onStateChanged(const truss::gui::state::WorkspaceState& newState) {
625 // Phase label
626 QString phaseText;
627 switch (newState.phase) {
628 case state::WorkspacePhase::Empty:
629 phaseText = QStringLiteral("Empty");
630 break;
631 case state::WorkspacePhase::ModelBuilding:
632 phaseText = QStringLiteral("Building");
633 break;
634 case state::WorkspacePhase::Validating:
635 phaseText = QStringLiteral("Validating…");
636 break;
637 case state::WorkspacePhase::Analysing:
638 phaseText = QStringLiteral("Analysing…");
639 break;
640 case state::WorkspacePhase::ResultsReady:
641 phaseText = QStringLiteral("Results Ready");
642 break;
643 }
644 m_phaseLabel->setText(phaseText);
645
646 // Stats label (node/member counts from models)
647 const int nodeCount = m_controller->nodeModel()->rowCount();

746// Event filter — centralised single-key shortcut dispatch
747// ============================================================
748
749bool MainWindow::eventFilter(QObject* /*watched*/, QEvent* event) {
750 if (event->type() != QEvent::KeyPress)
751 return false;
752
753 // Only intercept events that belong to this top-level window.
754 // This prevents the filter from interfering with modal dialogs
755 // (e.g. AnalysisOptionsDialog) that share the same qApp.
756 QWidget* fw = QApplication::focusWidget();
757 if (!fw || fw->window() != this)
758 return false;
759
760 auto* ke = static_cast<QKeyEvent*>(event);
761 const Qt::KeyboardModifiers mods = ke->modifiers();
762
763 // ── Ctrl-modified shortcuts (Ctrl = ⌘ on macOS) ─────────────────────────
764 // QAction::setShortcut(QKeySequence::New/Open/Save) plus Qt::WindowShortcut
765 // is theoretically sufficient, but child widgets (QTableView, QAbstractSpinBox)
766 // can respond to QEvent::ShortcutOverride and cause Qt to mark the shortcut as
767 // "ambiguous", silently dropping the dispatch. Intercepting here at the
768 // application level guarantees delivery regardless of focus placement.
769 //
770 // Returning true consumes the KeyPress so QShortcutMap never sees it,
771 // preventing double-firing. The menu shortcut hint text (Ctrl+N etc.) is
772 // preserved through the QAction::setShortcut() calls in setupMenuBar().
773 if (mods == Qt::ControlModifier) {
774 switch (ke->key()) {
775 case Qt::Key_N:
776 if (m_actNew && m_actNew->isEnabled()) {
777 m_actNew->trigger();
778 return true;
779 }
780 break;
781 case Qt::Key_O:
782 if (m_actOpen && m_actOpen->isEnabled()) {
783 m_actOpen->trigger();
784 return true;
785 }
786 break;
787 case Qt::Key_S:
788 if (m_actSave && m_actSave->isEnabled()) {
789 m_actSave->trigger();
790 return true;
791 }
792 break;
793 default:
794 break;
795 }
796 // All other Ctrl+key combos (Ctrl+Z, Ctrl+C, etc.) pass through.
797 return false;
798 }
799
800 // ── Bare single-key shortcuts ─────────────────────────────────────────────
801 // Modifier-combos (Ctrl+S/O/N) are handled above.
802 // Function keys (F5) remain on QAction::setShortcut() exclusively.
803 if (mods != Qt::NoModifier)
804 return false;
805
806 // Guard: leave the key for the focused text-input widget to process.
807 const bool inTextInput = isTextInputFocused();
808
809 switch (ke->key()) {
810 case Qt::Key_N:
811 if (!inTextInput && m_actToolNode && m_actToolNode->isEnabled()) {
812 m_actToolNode->trigger();
813 return true;
814 }
815 break;
816 case Qt::Key_M:
817 if (!inTextInput && m_actToolMember && m_actToolMember->isEnabled()) {
818 m_actToolMember->trigger();
819 return true;
820 }
821 break;
822 case Qt::Key_Escape:
823 if (!inTextInput && m_actToolSelect) {
824 m_actToolSelect->trigger();
825 return true;
826 }
827 break;
828 case Qt::Key_Z:
829 if (!inTextInput && m_actZoomFit) {
830 m_actZoomFit->trigger();
831 return true;
832 }
833 break;
834 case Qt::Key_Delete:
835 case Qt::Key_Backspace:
836 if (!inTextInput && m_canvas) {
837 m_canvas->triggerDeleteSelected();
838 return true;
839 }
840 break;
841 default:
842 break;
843 }
844 return false;
845}
