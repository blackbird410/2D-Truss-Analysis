150void TrussValidator::validateGeometry(const Truss& truss, ValidationResult& result) {
151 // Check for zero-length members
152 auto zeroLengthMembers = findZeroLengthMembers(truss, Constants::GEOMETRY_TOLERANCE);
153 for (auto memberId : zeroLengthMembers) {
154 auto member = truss.getMember(memberId);
155 if (member) {
156 ValidationIssue issue(ValidationSeverity::Error,
157 "Geometry",
158 "Member " + std::to_string(memberId) +
159 " has zero or near-zero length",
160 "Member length: " + std::to_string(member->getLength()) +
161 " m. "
162 "Zero-length members cannot carry loads and will cause "
163 "singular stiffness matrix.");
164 issue.affectedMembers.push_back(memberId);
165 issue.affectedNodes.push_back(member->getStartNode()->getId());
166 issue.affectedNodes.push_back(member->getEndNode()->getId());
167 result.addIssue(issue);
168 }
169 }
170
171 // Check for coincident nodes
172 auto coincidentPairs = findCoincidentNodePairs(truss, Constants::GEOMETRY_TOLERANCE);
173 for (const auto& [id1, id2] : coincidentPairs) {
174 ValidationIssue issue(ValidationSeverity::Warning,
175 "Geometry",
176 "Nodes " + std::to_string(id1) + " and " + std::to_string(id2) +
177 " are coincident or very close",
178 "Consider merging these nodes or checking for input errors.");
179 issue.affectedNodes = {id1, id2};
180 result.addIssue(issue);
181 }
182
183 // Check for duplicate members
184 auto duplicatePairs = findDuplicateMembers(truss);
185 for (const auto& [id1, id2] : duplicatePairs) {
186 ValidationIssue issue(
187 ValidationSeverity::Warning,
188 "Geometry",
189 "Members " + std::to_string(id1) + " and " + std::to_string(id2) +
190 " connect the same nodes",
191 "Duplicate members do not add structural stiffness in a truss model.");
192 issue.affectedMembers = {id1, id2};
193 result.addIssue(issue);
194 }
195
196 // Check for valid coordinates (no NaN or infinity)
197 for (const auto& node : truss.getNodes()) {
198 if (std::isnan(node->getX()) || std::isnan(node->getY()) || std::isinf(node->getX()) ||
199 std::isinf(node->getY())) {
200 ValidationIssue issue(ValidationSeverity::Fatal,
201 "Geometry",
202 "Node " + std::to_string(node->getId()) +
203 " has invalid coordinates (NaN or infinity)",
204 "Position: (" + std::to_string(node->getX()) + ", " +
205 std::to_string(node->getY()) + ")");
206 issue.affectedNodes.push_back(node->getId());
207 result.addIssue(issue);
208 }
209 }

212void TrussValidator::validateMaterials(const Truss& truss, ValidationResult& result) {
213 for (const auto& member : truss.getMembers()) {
214 const auto& material = member->getMaterial();
215 const auto& section = member->getSection();
216
217 // Check Young's modulus
218 if (material.youngModulus <= 0.0) {
219 ValidationIssue issue(ValidationSeverity::Error,
220 "Material",
221 "Member " + std::to_string(member->getId()) +
222 " has non-positive Young's modulus",
223 "Young's modulus: " + std::to_string(material.youngModulus) +
224 " Pa. Must be positive.");
225 issue.affectedMembers.push_back(member->getId());
226 result.addIssue(issue);
227 }
228
229 // Check cross-sectional area
230 if (section.area <= 0.0) {
231 ValidationIssue issue(ValidationSeverity::Error,
232 "Material",
233 "Member " + std::to_string(member->getId()) +
234 " has non-positive cross-sectional area",
235 "Area: " + std::to_string(section.area) +
236 " m². Must be positive.");
237 issue.affectedMembers.push_back(member->getId());
238 result.addIssue(issue);
239 }
240
241 // Check density (warning only, not critical for static analysis)
242 if (material.density <= 0.0) {
243 ValidationIssue issue(ValidationSeverity::Warning,
244 "Material",
245 "Member " + std::to_string(member->getId()) +
246 " has non-positive density",
247 "Density: " + std::to_string(material.density) +
248 " kg/m³. Self-weight calculations will be incorrect.");
249 issue.affectedMembers.push_back(member->getId());
250 result.addIssue(issue);
251 }
252
253 // Check yield strength (warning only)
254 if (material.yieldStrength <= 0.0) {
255 ValidationIssue issue(ValidationSeverity::Warning,
256 "Material",
257 "Member " + std::to_string(member->getId()) +
258 " has non-positive yield strength",
259 "Yield strength: " + std::to_string(material.yieldStrength) +

267void TrussValidator::validateBoundaryConditions(const Truss& truss, ValidationResult& result) {
268 // Check minimum constraints for stability (at least 3 constraints to prevent rigid body motion)
269 size_t totalConstraints = truss.getConstrainedDofs();
270
271 if (totalConstraints < 3) {
272 result.addIssue(ValidationIssue(
273 ValidationSeverity::Error,
274 "Boundary Conditions",
275 "Insufficient boundary constraints for stability",
276 "Total constraints: " + std::to_string(totalConstraints) +
277 ". "
278 "Minimum 3 constraints required to prevent rigid body motion (2D truss)."));
279 }
280
281 // Check if truss has any supports
282 auto constrainedNodes = truss.getConstrainedNodes();
283 if (constrainedNodes.empty()) {
284 result.addIssue(ValidationIssue(
285 ValidationSeverity::Error,
286 "Boundary Conditions",
287 "No support constraints defined",
288 "At least one node must have boundary conditions to prevent rigid body motion."));
289 }
290
291 // Check for proper constraint distribution
292 if (!checkRigidBodyStability(truss)) {
293 result.addIssue(ValidationIssue(ValidationSeverity::Error,
294 "Boundary Conditions",
295 "Boundary conditions do not prevent rigid body motion",
296 "The truss may rotate or translate as a rigid body. "
297 "Ensure supports are properly distributed and oriented."));
298 }
299
300 // Info: Report support configuration
301 size_t pinnedCount = 0, rollerCount = 0;
302 for (const auto& node : truss.getNodes()) {
303 if (node->isPinned())
304 pinnedCount++;
305 if (node->isRoller())
306 rollerCount++;
307 }
308
309 result.addIssue(ValidationIssue(ValidationSeverity::Info,
310 "Boundary Conditions",
311 "Support configuration: " + std::to_string(pinnedCount) +
312 " pinned, " + std::to_string(rollerCount) +
313 " roller supports",
314 "Total DOF constraints: " + std::to_string(totalConstraints)));
315}

317void TrussValidator::validateStaticDeterminacy(const Truss& truss, ValidationResult& result) {
318 size_t n = truss.getNodeCount();
319 size_t m = truss.getMemberCount();
320 size_t r = truss.getConstrainedDofs();
321
322 // Check equation: 2n = m + r
323 int determinacyCheck = static_cast<int>(2 * n) - static_cast<int>(m + r);
324
325 if (determinacyCheck == 0) {
326 result.addIssue(ValidationIssue(ValidationSeverity::Info,
327 "Static Determinacy",
328 "Truss is statically determinate",
329 "2n = m + r: 2×" + std::to_string(n) + " = " +
330 std::to_string(m) + " + " + std::to_string(r)));
331 } else if (determinacyCheck < 0) {
332 // 2n < m+r means too many members/constraints → indeterminate
333 result.addIssue(ValidationIssue(
334 ValidationSeverity::Warning,
335 "Static Determinacy",
336 "Truss is statically indeterminate (degree: " + std::to_string(-determinacyCheck) + ")",
337 "2n = " + std::to_string(2 * n) + ", m + r = " + std::to_string(m + r) +
338 ". "
339 "Indeterminate structures require advanced analysis methods."));
340 } else {
341 // 2n > m+r means too few members/constraints → unstable
342 result.addIssue(
343 ValidationIssue(ValidationSeverity::Error,
344 "Static Determinacy",
345 "Truss is unstable or has internal mechanisms (deficiency: " +
346 std::to_string(determinacyCheck) + ")",
347 "2n = " + std::to_string(2 * n) + ", m + r = " + std::to_string(m + r) +
348 ". "
349 "The structure lacks sufficient members or constraints."));
350 }

353void TrussValidator::validateKinematicStability(const Truss& truss, ValidationResult& result) {
354 // Critical stability check: prevent geometrically unstable structures from analysis
355
356 const auto& nodes = truss.getNodes();
357
358 // Check 1: Minimum constraint count
359 if (truss.getConstrainedDofs() < 3) {
360 result.addIssue(ValidationIssue(ValidationSeverity::Fatal,
361 "Kinematic Stability",
362 "Insufficient constraints: Structure has only " +
363 std::to_string(truss.getConstrainedDofs()) +
364 " constrained DOFs (minimum 3 required)",
365 "A 2D truss requires at least 3 constraints to prevent "
366 "rigid body motion (2 translations + 1 rotation)."));
367 return;
368 }
369
370 // Check 2: Rigid body mode prevention
371 bool hasXConstraint = false;
372 bool hasYConstraint = false;
373
374 for (const auto& node : nodes) {
375 SupportType support = node->getSupportType();
376
377 if (support == SupportType::Pinned || support == SupportType::RollerY) {
378 hasXConstraint = true;
379 }
380
381 if (support == SupportType::Pinned || support == SupportType::RollerX) {
382 hasYConstraint = true;
383 }
384 }
385
386 if (!hasXConstraint) {
387 result.addIssue(ValidationIssue(
388 ValidationSeverity::Fatal,
389 "Kinematic Stability",
390 "No horizontal constraints: Structure can translate freely in X direction",
391 "Add a support with X constraint (Pinned or RollerY)."));
392 }
393
394 if (!hasYConstraint) {
395 result.addIssue(ValidationIssue(
396 ValidationSeverity::Fatal,
397 "Kinematic Stability",
398 "No vertical constraints: Structure can translate freely in Y direction (GEOMETRIC "
399 "INSTABILITY)",
400 "CRITICAL: This causes singular stiffness matrix and astronomical displacements. "
401 "For typical bridge truss: Left=Pinned, Right=RollerX (NOT RollerY). "
402 "RollerY allows vertical movement and makes the structure unstable under vertical "
403 "loads."));
404 }
405
406 if (!hasXConstraint || !hasYConstraint) {
407 return; // Don't continue if fatal errors found
408 }
409
410 // Check 3: Full stability check
411 if (!checkMinimumConstraints(truss)) {
412 result.addIssue(
413 ValidationIssue(ValidationSeverity::Error,
414 "Kinematic Stability",
415 "Structure lacks adequate constraints for kinematic stability",
416 "The structure may contain mechanisms. Verify support configuration "
417 "prevents all rigid body modes."));
418 }
419
420 // Check 4: Isolated nodes
421 for (const auto& node : nodes) {
422 auto connectedMembers = truss.getMembersConnectedTo(node->getId());
423 if (connectedMembers.empty() && !node->isConstrained()) {
424 ValidationIssue issue(
425 ValidationSeverity::Warning,
426 "Kinematic Stability",
427 "Node " + std::to_string(node->getId()) +
428 " is not connected to any member and has no support",
429 "This node is isolated and will not participate in load transfer.");
430 issue.affectedNodes.push_back(node->getId());
431 result.addIssue(issue);
432 }
433 }

436void TrussValidator::validateLoads(const Truss& truss, ValidationResult& result) {
437 // Check if any loads are applied
438 if (!truss.hasAppliedForces()) {
439 result.addIssue(ValidationIssue(
440 ValidationSeverity::Warning,
441 "Loads",
442 "No external forces applied to the truss",
443 "Analysis will compute only support reactions with zero member forces."));
444 return;
445 }
446
447 // Check each loaded node
448 size_t totalLoads = 0;
449 for (const auto& node : truss.getNodes()) {
450 if (node->hasAppliedForce()) {
451 totalLoads++;
452
453 // Check for NaN or infinity in forces
454 const auto& force = node->getAppliedForce();
455 if (std::isnan(force.fx) || std::isnan(force.fy) || std::isinf(force.fx) ||
456 std::isinf(force.fy)) {
457 ValidationIssue issue(ValidationSeverity::Error,
458 "Loads",
459 "Node " + std::to_string(node->getId()) +
460 " has invalid force values (NaN or infinity)",
461 "Force: (" + std::to_string(force.fx) + ", " +
462 std::to_string(force.fy) + ")");
463 issue.affectedNodes.push_back(node->getId());
464 result.addIssue(issue);
465 }
466
467 // Warning for loads on fully constrained nodes
468 if (node->getDegreesOfFreedom() == 0) {
469 ValidationIssue issue(ValidationSeverity::Warning,
470 "Loads",
471 "Force applied to fully constrained node " +
472 std::to_string(node->getId()),
473 "This force will be entirely resisted by support reactions "
474 "and not transferred through members.");
475 issue.affectedNodes.push_back(node->getId());
476 result.addIssue(issue);
477 }
478 }
479 }
480
481 result.addIssue(ValidationIssue(ValidationSeverity::Info,
482 "Loads",
483 std::to_string(totalLoads) + " node(s) have applied forces",
484 ""));
485}

487void TrussValidator::validateConnectivity(const Truss& truss, ValidationResult& result) {
488 // Check each member for valid node connections
489 for (const auto& member : truss.getMembers()) {
490 auto startNode = member->getStartNode();
491 auto endNode = member->getEndNode();
492
493 // Check for null node pointers
494 if (!startNode || !endNode) {
495 ValidationIssue issue(ValidationSeverity::Fatal,
496 "Connectivity",
497 "Member " + std::to_string(member->getId()) +
498 " has null node connection(s)",
499 "Internal error: Member contains null node pointer.");
500 issue.affectedMembers.push_back(member->getId());
501 result.addIssue(issue);
502 continue;
503 }
504
505 // Check for self-loops (member connecting node to itself)
506 if (startNode->getId() == endNode->getId()) {
507 ValidationIssue issue(ValidationSeverity::Error,
508 "Connectivity",
509 "Member " + std::to_string(member->getId()) + " connects node " +
510 std::to_string(startNode->getId()) + " to itself",
511 "Self-loop members are not valid in truss structures.");
512 issue.affectedMembers.push_back(member->getId());
513 issue.affectedNodes.push_back(startNode->getId());
514 result.addIssue(issue);
515 }
516
517 // Verify nodes exist in truss
518 bool startExists = truss.getNode(startNode->getId()) != nullptr;
519 bool endExists = truss.getNode(endNode->getId()) != nullptr;
520
521 if (!startExists || !endExists) {
522 ValidationIssue issue(
523 ValidationSeverity::Fatal,
524 "Connectivity",
525 "Member " + std::to_string(member->getId()) + " references non-existent node(s)",
526 "Start node exists: " + std::string(startExists ? "yes" : "no") +
527 ", End node exists: " + std::string(endExists ? "yes" : "no"));
528 issue.affectedMembers.push_back(member->getId());
529 result.addIssue(issue);
530 }
531 }

536bool TrussValidator::checkMinimumConstraints(const Truss& truss) {
537 // For 2D truss: minimum 3 constraints needed to prevent rigid body motion
538 // BUT: must check that constraints prevent ALL 3 rigid body modes:
539 // - Horizontal translation (need at least 1 X constraint)
540 // - Vertical translation (need at least 1 Y constraint)
541 // - Rotation (need non-collinear constraints)
542
543 if (truss.getConstrainedDofs() < 3) {
544 return false;
545 }
546
547 // Check rigid body stability
548 return checkRigidBodyStability(truss);
549}

551bool TrussValidator::checkRigidBodyStability(const Truss& truss) {
552 // Rigid body stability in 2D requires preventing:
553 // 1. Horizontal translation (X direction)
554 // 2. Vertical translation (Y direction)
555 // 3. Rotation about any point
556
557 const auto& nodes = truss.getNodes();
558
559 // Count constraints by type
560 bool hasXConstraint = false; // At least one fixed X DOF
561 bool hasYConstraint = false; // At least one fixed Y DOF
562 int xConstraintCount = 0;
563 int yConstraintCount = 0;
564
565 for (const auto& node : nodes) {
566 SupportType support = node->getSupportType();
567
568 // Check X constraints (horizontal restraint)
569 if (support == SupportType::Pinned || support == SupportType::RollerY) {
570 hasXConstraint = true;
571 xConstraintCount++;
572 }
573
574 // Check Y constraints (vertical restraint)
575 if (support == SupportType::Pinned || support == SupportType::RollerX) {
576 hasYConstraint = true;
577 yConstraintCount++;
578 }
579 }
580
581 // CRITICAL: Must have at least one constraint in EACH direction
582 if (!hasXConstraint || !hasYConstraint) {
583 return false; // Unstable: can translate as rigid body
584 }
585
586 // Additional check: need sufficient constraints to prevent rotation
587 // Minimum 3 total constraints required
588 if (xConstraintCount + yConstraintCount < 3) {
589 return false;
590 }
591
592 // For proper rotational stability, constraints must be non-collinear
593 // If all Y constraints are at same X coordinate, structure is unstable
594 // This is a more advanced check - for now, basic count is sufficient
595
596 return true;
597}

611std::vector<std::pair<NodeId, NodeId>> TrussValidator::findCoincidentNodePairs(const Truss& truss,
612 Real tolerance) {
613 std::vector<std::pair<NodeId, NodeId>> pairs;
614 const auto& nodes = truss.getNodes();
615
616 for (size_t i = 0; i < nodes.size(); ++i) {
617 for (size_t j = i + 1; j < nodes.size(); ++j) {
618 if (nodes[i]->distanceTo(*nodes[j]) < tolerance) {
619 pairs.emplace_back(nodes[i]->getId(), nodes[j]->getId());
620 }
621 }
622 }
623
624 return pairs;
625}

639std::vector<std::pair<MemberId, MemberId>>
640TrussValidator::findDuplicateMembers(const Truss& truss) {
641 std::vector<std::pair<MemberId, MemberId>> duplicates;
642 const auto& members = truss.getMembers();
643
644 for (size_t i = 0; i < members.size(); ++i) {
645 for (size_t j = i + 1; j < members.size(); ++j) {
646 NodeId start1 = members[i]->getStartNode()->getId();
647 NodeId end1 = members[i]->getEndNode()->getId();
648 NodeId start2 = members[j]->getStartNode()->getId();
649 NodeId end2 = members[j]->getEndNode()->getId();
650
651 // Check if they connect the same nodes (in either direction)
652 if ((start1 == start2 && end1 == end2) || (start1 == end2 && end1 == start2)) {
653 duplicates.emplace_back(members[i]->getId(), members[j]->getId());
654 }
655 }
656 }
657
658 return duplicates;
659}
