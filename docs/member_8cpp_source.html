166bool Member::intersectsWith(const Member& other, Real tolerance) const {
167 // Get line segments
168 Point2D p1 = m_startNode->getPosition();
169 Point2D q1 = m_endNode->getPosition();
170 Point2D p2 = other.m_startNode->getPosition();
171 Point2D q2 = other.m_endNode->getPosition();
172
173 // Check if lines are collinear or parallel
174 Vector2d dir1 = getDirection();
175 Vector2d dir2 = other.getDirection();
176 Real crossProduct = (dir1.x() * dir2.y()) - (dir1.y() * dir2.x());
177
178 if (truss::utils::math::isZero(crossProduct, tolerance)) {
179 // Lines are parallel or collinear
180 return false;
181 }
182
183 // Calculate intersection parameters
184 Real dx1 = q1.x - p1.x;
185 Real dy1 = q1.y - p1.y;
186 Real dx2 = q2.x - p2.x;
187 Real dy2 = q2.y - p2.y;
188 Real dx12 = p1.x - p2.x;
189 Real dy12 = p1.y - p2.y;
190
191 Real denom = (dx1 * dy2) - (dy1 * dx2);
192 if (truss::utils::math::isZero(denom, tolerance)) {
193 return false;
194 }
195
196 Real t1 = ((dx2 * dy12) - (dy2 * dx12)) / denom;
197 Real t2 = ((dx1 * dy12) - (dy1 * dx12)) / denom;
198
199 // Check if intersection point lies within both line segments
200 return (t1 >= -tolerance && t1 <= 1.0 + tolerance && t2 >= -tolerance && t2 <= 1.0 + tolerance);
201}

203Point2D Member::getIntersectionPoint(const Member& other) const {
204 // Get line segments
205 Point2D p1 = m_startNode->getPosition();
206 Point2D q1 = m_endNode->getPosition();
207 Point2D p2 = other.m_startNode->getPosition();
208 Point2D q2 = other.m_endNode->getPosition();
209
210 // Calculate intersection using parametric line equations
211 Real dx1 = q1.x - p1.x;
212 Real dy1 = q1.y - p1.y;
213 Real dx2 = q2.x - p2.x;
214 Real dy2 = q2.y - p2.y;
215 Real dx12 = p1.x - p2.x;
216 Real dy12 = p1.y - p2.y;
217
218 Real denom = (dx1 * dy2) - (dy1 * dx2);
219 if (truss::utils::math::isZero(denom)) {
220 // Lines are parallel or collinear - no unique intersection
221 throw std::runtime_error(
222 "Cannot calculate intersection point: lines are parallel or collinear");
223 }
224
225 Real t1 = ((dx2 * dy12) - (dy2 * dx12)) / denom;
226
227 // Calculate intersection point using parameter t1
228 Point2D intersection;
229 intersection.x = p1.x + (t1 * dx1);
230 intersection.y = p1.y + (t1 * dy1);
231
232 return intersection;
233}

258MatrixXd Member::getGlobalStiffnessMatrix() const {
259 MatrixXd local = getLocalStiffnessMatrix();
260
261 // Create full transformation matrix
262 Vector2d unitVec = getUnitVector();
263 Real c = unitVec.x();
264 Real s = unitVec.y();
265
266 MatrixXd T(4, 4);
267 T << c, s, 0, 0, -s, c, 0, 0, 0, 0, c, s, 0, 0, -s, c;
