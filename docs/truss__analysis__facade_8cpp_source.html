169AnalysisWorkflowResult
170TrussAnalysisFacade::analyzeFromFile(const std::filesystem::path& filepath,
171 const core::analysis::AnalysisOptions& options) {
172 // Step 1: Load truss from file
173 auto loadResult = loadTruss(filepath);
174 if (!loadResult) {
175 return AnalysisWorkflowResult::Failure("Failed to load truss: " + loadResult.errorMessage);
176 }
177 auto trussHandle = loadResult.value;
178
179 // Step 2: Validate truss structure
180 auto validationResult = validateTruss(trussHandle);
181 if (!validationResult || !validationResult.value.isValid()) {
182 std::string error = validationResult ? formatValidationErrors(validationResult.value)
183 : validationResult.errorMessage;
184 clearTruss(trussHandle);
185 return AnalysisWorkflowResult::Failure("Validation failed: " + error);
186 }
187
188 // Step 3: Perform analysis
189 auto& truss = getTrussMutable(trussHandle);
190 auto analysisResult = analyze(truss, options);
191 if (!analysisResult) {
192 clearTruss(trussHandle);
193 return AnalysisWorkflowResult::Failure("Analysis failed: " + analysisResult.errorMessage);
194 }
195
196 return AnalysisWorkflowResult::Success(trussHandle, analysisResult.value);
197}

199AnalysisWorkflowResult
200TrussAnalysisFacade::analyzeInteractive(TrussBuilder& builder,
201 const core::analysis::AnalysisOptions& options) {
202 // Step 1: Build truss from builder
203 std::shared_ptr<core::Truss> trussPtr;
204 try {
205 trussPtr = builder.build();
206 } catch (const std::exception& e) {
207 return AnalysisWorkflowResult::Failure(std::string("Failed to build truss: ") + e.what());
208 }
209
210 // Step 2: Create truss handle
211 auto createResult = createTruss("Interactive Truss");
212 if (!createResult) {
213 return AnalysisWorkflowResult::Failure("Failed to create truss: " +
214 createResult.errorMessage);
215 }
216 auto trussHandle = createResult.value;
217
218 // Copy built truss into managed truss
219 auto& managedTruss = getTrussMutable(trussHandle);
220 managedTruss = *trussPtr;
221
222 // Step 3: Validate
223 auto validationResult = validateTruss(trussHandle);
224 if (!validationResult || !validationResult.value.isValid()) {
225 std::string error = validationResult ? formatValidationErrors(validationResult.value)
226 : validationResult.errorMessage;
227 clearTruss(trussHandle);
228 return AnalysisWorkflowResult::Failure("Validation failed: " + error);
229 }
230
231 // Step 4: Analyze
232 auto analysisResult = analyze(managedTruss, options);
233 if (!analysisResult) {
234 clearTruss(trussHandle);
235 return AnalysisWorkflowResult::Failure("Analysis failed: " + analysisResult.errorMessage);
236 }
237
238 return AnalysisWorkflowResult::Success(trussHandle, analysisResult.value);
239}

241core::validation::ValidationResult
242TrussAnalysisFacade::validateFromFile(const std::filesystem::path& filepath) {
243 // Load truss
244 auto loadResult = loadTruss(filepath);
245 if (!loadResult) {
246 core::validation::ValidationResult result;
247 core::validation::ValidationIssue issue(core::validation::ValidationSeverity::Fatal,
248 "File Loading",
249 "Failed to load truss file: " +
250 loadResult.errorMessage,
251 "Check file path and format");
252 result.addIssue(issue);
253 return result;
254 }
255
256 auto trussHandle = loadResult.value;
257
258 // Validate
259 auto validationResult = validateTruss(trussHandle);
260
261 // Cleanup and return
262 clearTruss(trussHandle);
263
264 if (validationResult) {
265 return validationResult.value;
266 }
267 core::validation::ValidationResult result;
268 core::validation::ValidationIssue issue(core::validation::ValidationSeverity::Fatal,
269 "Validation",
270 "Validation process failed: " +
271 validationResult.errorMessage,
272 "Internal error during validation");
273 result.addIssue(issue);
274 return result;
275}

277core::validation::ValidationResult TrussAnalysisFacade::validateBuilder(TrussBuilder& builder) {
278 std::shared_ptr<core::Truss> trussPtr;
279 try {

282 core::validation::ValidationResult result;

284 "Build Error",

395std::string TrussAnalysisFacade::formatValidationErrors(
396 const core::validation::ValidationResult& result) const {
397 std::ostringstream oss;
398 oss << result.getSummary() << "\n";
399 auto errors = result.getErrorMessages();
400 if (!errors.empty()) {
401 oss << "\nErrors:\n";
402 for (const auto& error : errors) {
403 oss << " • " << error << "\n";
404 }
405 }
406 auto warnings = result.getWarningMessages();
407 if (!warnings.empty()) {
408 oss << "\nWarnings:\n";
409 for (size_t i = 0; i < std::min(warnings.size(), static_cast<size_t>(5)); ++i) {
410 oss << " • " << warnings.at(i) << "\n";
411 }
412 if (warnings.size() > 5) {
413 oss << " ... and " << (warnings.size() - 5) << " more warnings\n";
414 }
415 }
416 return oss.str();
417}
