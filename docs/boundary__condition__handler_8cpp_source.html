15std::vector<Index> BoundaryConditionHandler::getFreeDofs(const Truss& truss) {
16 std::vector<Index> freeDofs;
17 const auto& nodes = truss.getNodes();
18
19 for (const auto& node : nodes) {
20 Index dofX = node->getDofX();
21 Index dofY = node->getDofY();
22 SupportType support = node->getSupportType();
23
24 // Add free DOFs based on support type
25 // Pinned: Both DOFs constrained (neither free)
26 // RollerX: X free, Y constrained
27 // RollerY: Y free, X constrained
28 // Free: Both DOFs free
29 if (support == SupportType::Free || support == SupportType::RollerX) {
30 freeDofs.push_back(dofX); // X DOF is free
31 }
32
33 if (support == SupportType::Free || support == SupportType::RollerY) {
34 freeDofs.push_back(dofY); // Y DOF is free
35 }
36 }
37
38 // Sort the indices for consistent ordering
39 std::sort(freeDofs.begin(), freeDofs.end());
40
41 return freeDofs;
42}

44std::vector<Index> BoundaryConditionHandler::getConstrainedDofs(const Truss& truss) {
45 std::vector<Index> constrainedDofs;

47

50 Index dofY = node->getDofY();
51 SupportType support = node->getSupportType();
52
53 // Add constrained DOFs based on support type
54 // Pinned: Both X and Y constrained (2 reactions)
55 // RollerX: Only Y constrained (1 reaction)
56 // RollerY: Only X constrained (1 reaction)

58 constrainedDofs.push_back(dofX); // X DOF is constrained
59 }
60
61 if (support == SupportType::Pinned || support == SupportType::RollerX) {
62 constrainedDofs.push_back(dofY); // Y DOF is constrained
63 }
64 }
65
66 // Sort the indices for consistent ordering
67 std::sort(constrainedDofs.begin(), constrainedDofs.end());
