24int ExampleCommand::execute() {

40 .applyForce(NodeId(3), 0.0, -50000.0); // 50 kN downward load

42 if (m_verbose) {
43 m_presenter.displayInfo("Truss structure created using fluent builder API.\n");
44 }
45
46 // Analyze using facade
47 m_presenter.displayInfo("Performing structural analysis...\n");
48 auto analysisResult = m_facade.analyzeInteractive(builder);
49
50 if (!analysisResult) {
51 m_presenter.displayError("Analysis failed: " + analysisResult.errorMessage);
52 return 1;
53 }

55 if (m_verbose) {
56 m_presenter.displaySuccess("Analysis completed successfully.");
