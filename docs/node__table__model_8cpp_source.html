76QVariant NodeTableModel::data(const QModelIndex& index, int role) const {
77 if (!index.isValid())
78 return {};
79
80 const int row = index.row();
81 const int col = index.column();
82
83 if (row < 0 || row >= static_cast<int>(m_rows.size()))
84 return {};
85 if (col < 0 || col >= kColumnCount)
86 return {};
87
88 const NodeView& n = m_rows[static_cast<size_t>(row)];
89
90 // ------------------------------------------------------------------
91 // TextAlignmentRole
92 // ------------------------------------------------------------------
93 if (role == Qt::TextAlignmentRole) {
94 if (col == kColId || col == kColSupport)
95 return static_cast<int>(Qt::AlignCenter);
96 return static_cast<int>(Qt::AlignRight | Qt::AlignVCenter);
97 }
98
99 // ------------------------------------------------------------------
100 // BackgroundRole — amber tint for loaded nodes
101 // ------------------------------------------------------------------
102 if (role == Qt::BackgroundRole) {
103 if (isLoaded(n))
104 return kLoadedBg;
105 return {};
106 }
107
108 // ------------------------------------------------------------------
109 // ToolTipRole — support description (col 3) + numeric precision elsewhere
110 // ------------------------------------------------------------------
111 if (role == Qt::ToolTipRole) {
112 if (col == kColSupport)
113 return formatSupport(n.support);
114 return {};
115 }
116
117 // ------------------------------------------------------------------
118 // DisplayRole
119 // ------------------------------------------------------------------
120 if (role != Qt::DisplayRole)
121 return {};
122
123 // Results columns: show "—" when no analysis has been run
124 const bool isResultsCol = (col >= kColDx && col <= kColRy);
125 if (isResultsCol && !m_hasResults)
126 return QStringLiteral("—");
127
128 switch (col) {
129 case kColId:
130 return QString::number(n.id);
131
132 case kColX:
133 return QString::number(n.x, 'f', 4);
134
135 case kColY:
136 return QString::number(n.y, 'f', 4);
137
138 case kColSupport:
139 return formatSupport(n.support);
140
141 case kColFx:
142 // Convert N → kN
143 return QString::number(n.fx / 1000.0, 'f', 3);
144
145 case kColFy:
146 return QString::number(n.fy / 1000.0, 'f', 3);
147
148 case kColDx:
149 // Convert m → mm for readability
150 return QString::number(n.dx * 1000.0, 'f', 4);
151
152 case kColDy:
153 return QString::number(n.dy * 1000.0, 'f', 4);
154
155 case kColRx:
