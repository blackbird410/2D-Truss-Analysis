97void AnalysisControlBar::onStateChanged(const truss::gui::state::WorkspaceState& state) {
98 using truss::gui::state::WorkspacePhase;
99
100 switch (state.phase) {
101 case WorkspacePhase::Analysing:
102 m_actionStack->setCurrentIndex(1); // show progress bar
103 m_validateBtn->setEnabled(false);
104 m_optionsBtn->setEnabled(false);
105 break;
106
107 case WorkspacePhase::ModelBuilding:
108 case WorkspacePhase::ResultsReady:
109 m_actionStack->setCurrentIndex(0); // show run button
110 m_runBtn->setEnabled(state.hasTruss());
111 m_validateBtn->setEnabled(state.hasTruss());
112 m_optionsBtn->setEnabled(true);
113 break;
114
115 case WorkspacePhase::Empty:
116 case WorkspacePhase::Validating:
117 default:
118 m_actionStack->setCurrentIndex(0);
119 m_runBtn->setEnabled(false);
120 m_validateBtn->setEnabled(false);
121 m_optionsBtn->setEnabled(true);
122 break;
123 }
124}
