20std::unique_ptr<ITrussReader> FileIOFactory::createReader(FileFormat format) {
21 switch (format) {
22 case FileFormat::JSON:
23 return std::make_unique<JsonTrussReader>();
24
25 case FileFormat::XML:
26 return std::make_unique<XmlTrussReader>();

28 case FileFormat::Auto:
29 throw std::invalid_argument("Cannot create reader with Auto format. "

34 }
