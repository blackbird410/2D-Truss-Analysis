26int AnalyzeCommand::execute() {
27 m_presenter.displayHeader();
28 m_presenter.displayInfo("Analyzing truss structure from file...\n");
29
30 // Validate input file
31 if (!validateInputFile()) {
32 m_presenter.displayError("Input file does not exist or is not readable: " + m_inputFile);
33 m_presenter.displayInfo("Suggestion: Check the file path and ensure the file exists.\n");
34 return 1;
35 }
36
37 if (m_verbose) {
38 m_presenter.displayInfo("Input file: " + m_inputFile + "\n");
39 }
40
41 // ===== SINGLE FACADE CALL REPLACES 5+ SERVICE CALLS =====
42 // Old approach: loadTruss(), getTrussView(), getTrussMutable(),
43 // validateTruss(), analyze()
44 // New approach: analyzeFromFile() does all of the above
45 auto analysisResult = m_facade.analyzeFromFile(m_inputFile);
46
47 if (!analysisResult) {
48 m_presenter.displayError("Analysis failed: " + analysisResult.errorMessage);
49 m_presenter.displayInfo(
50 "Suggestion: Check the file format and ensure the structure is valid.\n");
51 return 1;
52 }
53
54 if (m_verbose) {
55 m_presenter.displaySuccess("Analysis completed successfully.");
56 }
57
58 // Display truss statistics
59 const auto& trussView = m_facade.getTrussView(analysisResult.trussHandle);
60 m_presenter.displayTrussStatistics(trussView);
61
62 // Display results
63 const auto& resultsView = m_facade.getResultsView(analysisResult.resultsHandle);
64 m_presenter.displayAnalysisResults(resultsView);
65
66 // Export results if requested
67 if (m_outputFile.has_value()) {
68 ExportFormat format;
69
70 // Determine format
71 if (m_exportFormat.has_value()) {
72 auto parsedFormat = parseExportFormat(m_exportFormat.value());
73 if (!parsedFormat.has_value()) {
74 m_presenter.displayError("Invalid export format: " + m_exportFormat.value());
75 m_presenter.displayInfo("Valid formats: JSON, XML, CSV, TSV, TXT, LaTeX, HTML\n");
76 return 1;
77 }
78 format = parsedFormat.value();
79 } else {
80 format = getDefaultExportFormat(m_outputFile.value());
81 }
82
83 if (m_verbose) {
84 m_presenter.displayInfo("\nExporting results to: " + m_outputFile.value() + "\n");
85 }
86
87 // Use facade export method
88 if (!m_facade.exportResults(analysisResult.resultsHandle, format, m_outputFile.value())) {
89 m_presenter.displayError("Export failed: Check output directory is writable.");
90 m_presenter.displayInfo("Suggestion: Verify that the output directory exists and you "
91 "have write permissions.\n");
92 return 1;
93 }
94
95 m_presenter.displaySuccess("Results exported to " + m_outputFile.value());
96 }
97
98 // Automatic cleanup via RAII (resources held by facade)
99 // No manual clearResults() or clearTruss() needed
100
101 m_presenter.displaySuccess("\nAnalysis complete!");
102 return 0;
103}

113bool AnalyzeCommand::validateInputFile() const {
114 namespace fs = std::filesystem;
115
116 try {
117 if (!fs::exists(m_inputFile)) {
118 return false;
119 }
120
121 if (!fs::is_regular_file(m_inputFile)) {
122 return false;
123 }
124
125 // Check read permissions
126 auto perms = fs::status(m_inputFile).permissions();
127 return (perms & fs::perms::owner_read) != fs::perms::none;
128
129 } catch (const std::filesystem::filesystem_error&) {
130 return false;
131 }
132}

134std::optional<truss::ExportFormat> AnalyzeCommand::parseExportFormat(const std::string& formatStr) {
135 using namespace truss::infrastructure::export_;
136
137 // Convert to uppercase for case-insensitive comparison
138 std::string upper = formatStr;
139 std::transform(
140 upper.begin(), upper.end(), upper.begin(), [](unsigned char c) { return std::toupper(c); });
141
142 if (upper == "JSON")
143 return ExportFormat::JSON;
144 if (upper == "XML")
145 return ExportFormat::XML;
146 if (upper == "CSV")
147 return ExportFormat::CSV;
148 if (upper == "TSV")
149 return ExportFormat::TSV;
150 if (upper == "TXT")
151 return ExportFormat::TXT;
152 if (upper == "LATEX" || upper == "TEX")
153 return ExportFormat::LaTeX;
154 if (upper == "HTML")
155 return ExportFormat::HTML;
156
157 return std::nullopt;
158}

160truss::ExportFormat AnalyzeCommand::getDefaultExportFormat(const std::string& filepath) {
161 using namespace truss::infrastructure::export_;
162
163 namespace fs = std::filesystem;
164 std::string ext = fs::path(filepath).extension().string();
165
166 // Convert to lowercase
167 std::transform(
168 ext.begin(), ext.end(), ext.begin(), [](unsigned char c) { return std::tolower(c); });
169
170 if (ext == ".json")
171 return ExportFormat::JSON;
172 if (ext == ".xml")
173 return ExportFormat::XML;
174 if (ext == ".csv")
175 return ExportFormat::CSV;
176 if (ext == ".tsv")
177 return ExportFormat::TSV;
178 if (ext == ".txt")
179 return ExportFormat::TXT;
180 if (ext == ".tex")
181 return ExportFormat::LaTeX;
182 if (ext == ".html")
183 return ExportFormat::HTML;
184
185 // Default to JSON
186 return ExportFormat::JSON;
187}
