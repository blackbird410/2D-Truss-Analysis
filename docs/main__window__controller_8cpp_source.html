168void MainWindowController::onTrussModified(std::size_t trussHandle) {
169 state::WorkspaceState next = m_state;
170 next.trussHandle = trussHandle;
171 next.phase = state::WorkspacePhase::ModelBuilding;
172 next.resultsHandle = 0;
173 next.isDirty = true;
174 next.lastError.clear();
175
176 // New model invalidates previous analysis results
177 emit resultsViewChanged(nullptr);
178
179 // Refresh node and member models from the new truss view
180 const core::interfaces::ITrussView* viewPtr = nullptr;
181 try {
182 if (trussHandle != 0) {
183 const auto& view = m_facade->getTrussView(trussHandle);
184 m_nodeModel->refresh(view);
185 m_nodeModel->setHasResults(false);
186 m_memberModel->refresh(view);
187 m_memberModel->setHasResults(false);
188 viewPtr = &view;
189
190 // Run validation and update validation model
191 auto valResult = m_facade->validateTruss(trussHandle);
192 if (valResult.success) {
193 m_validationModel->refresh(valResult.value);
194 }
195 }
196 } catch (...) {
197 // getTrussView may throw for invalid handles — silently ignore
198 }
199
200 emit trussViewChanged(viewPtr);
201 setState(std::move(next));
202}

211void MainWindowController::onAnalysisCompleted(std::size_t resultsHandle) {
212 state::WorkspaceState next = m_state;
213 next.resultsHandle = resultsHandle;
214 next.phase = state::WorkspacePhase::ResultsReady;
215 next.lastError.clear();
216
217 // Refresh results model and enable results columns in node/member models
218 try {
219 if (resultsHandle != 0) {
220 const auto& results = m_facade->getResultsView(resultsHandle);
221 m_resultsModel->refresh(results);
222 emit resultsViewChanged(&results); // forward to ResultsDockPanel
223 }
224 if (next.trussHandle != 0) {
225 const auto& view = m_facade->getTrussView(next.trussHandle);
226 m_nodeModel->refresh(view);
227 m_nodeModel->setHasResults(true);
228 m_memberModel->refresh(view);
229 m_memberModel->setHasResults(true);
230 // Forward the post-analysis view to the canvas. This view now contains
231 // nodal displacements (dx, dy) and reaction forces (rx, ry) which are
232 // required for DeformedShape rendering and force/reaction arrows.
233 emit trussViewChanged(&view);
234 }
235 } catch (...) {
236 // Silently ignore refresh failures (shouldn't happen with valid handles)
237 }
238
239 setState(std::move(next));
240}

242void MainWindowController::onAnalysisFailed(const QString& errorMessage) {
243 state::WorkspaceState next = m_state;
244 next.phase = state::WorkspacePhase::ModelBuilding;
245 next.resultsHandle = 0;
246 next.lastError = errorMessage.toStdString();
247 emit resultsViewChanged(nullptr); // invalidate stiffness matrix tab
248 setState(std::move(next));
249}
