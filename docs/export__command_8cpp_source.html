28 using namespace truss::infrastructure::export_;
29
30 m_presenter.displayHeader();

32
33 // Validate input files
34 if (!validateInputFile(m_trussFile)) {

39
40 if (m_verbose) {
41 m_presenter.displayInfo("Truss file: " + m_trussFile + "\n");
42 m_presenter.displayInfo("Output file: " + m_outputFile + "\n");
43 }
44
45 // Analyze truss from file using facade
46 m_presenter.displayInfo("\nAnalyzing truss to generate results for export...\n");
47 auto analysisResult = m_facade.analyzeFromFile(m_trussFile);

49 if (!analysisResult) {
50 m_presenter.displayError("Analysis failed: " + analysisResult.errorMessage);
51 m_presenter.displayInfo("Suggestion: Validate the truss structure first.\n");
52 m_presenter.displayInfo("Example: TrussAnalysisCLI validate --file " + m_trussFile + "\n");

88 break;

91 break;

135bool ExportCommand::validateInputFile(const std::string& filepath) {
136 namespace fs = std::filesystem;
137
138 try {
139 if (!fs::exists(filepath)) {
140 return false;
141 }
142
143 if (!fs::is_regular_file(filepath)) {
144 return false;
145 }
146
147 // Check read permissions
148 auto perms = fs::status(filepath).permissions();
149 return (perms & fs::perms::owner_read) != fs::perms::none;
150
151 } catch (const std::filesystem::filesystem_error&) {
152 return false;
153 }
154}

156std::optional<truss::ExportFormat> ExportCommand::parseExportFormat(const std::string& formatStr) {
157 using namespace truss::infrastructure::export_;
158
159 // Convert to uppercase for case-insensitive comparison
160 std::string upper = formatStr;
161 std::transform(
162 upper.begin(), upper.end(), upper.begin(), [](unsigned char c) { return std::toupper(c); });
163
164 if (upper == "JSON")
165 return ExportFormat::JSON;
166 if (upper == "XML")
167 return ExportFormat::XML;
168 if (upper == "CSV")
169 return ExportFormat::CSV;
170 if (upper == "TSV")
171 return ExportFormat::TSV;
172 if (upper == "TXT")
173 return ExportFormat::TXT;
174 if (upper == "LATEX" || upper == "TEX")
175 return ExportFormat::LaTeX;
176 if (upper == "HTML")
177 return ExportFormat::HTML;
178
179 return std::nullopt;
180}

182truss::ExportFormat ExportCommand::getDefaultExportFormat(const std::string& filepath) {
183 using namespace truss::infrastructure::export_;
184
185 namespace fs = std::filesystem;
186 std::string ext = fs::path(filepath).extension().string();
187
188 // Convert to lowercase
189 std::transform(
190 ext.begin(), ext.end(), ext.begin(), [](unsigned char c) { return std::tolower(c); });
191
192 if (ext == ".json")
193 return ExportFormat::JSON;
194 if (ext == ".xml")
195 return ExportFormat::XML;
196 if (ext == ".csv")
197 return ExportFormat::CSV;
198 if (ext == ".tsv")
199 return ExportFormat::TSV;
200 if (ext == ".txt")
201 return ExportFormat::TXT;
202 if (ext == ".tex")
203 return ExportFormat::LaTeX;
204 if (ext == ".html")
205 return ExportFormat::HTML;
206
207 // Default to JSON
208 return ExportFormat::JSON;
209}
