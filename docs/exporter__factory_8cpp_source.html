49ExportFormat ExporterFactory::detectFormat(const std::filesystem::path& filePath) {
50 std::string extension = filePath.extension().string();
51
52 // Convert to lowercase for case-insensitive comparison
53 extension = truss::utils::string::toLower(extension);
54
55 if (extension == ".csv") {
56 return ExportFormat::CSV;
57 } else if (extension == ".tsv" || extension == ".tab") {
58 return ExportFormat::TSV;
59 } else if (extension == ".json") {
60 return ExportFormat::JSON;
61 } else if (extension == ".xml") {
62 return ExportFormat::XML;
63 } else if (extension == ".html" || extension == ".htm") {
64 return ExportFormat::HTML;
65 } else if (extension == ".txt") {
66 return ExportFormat::TXT;
67 } else if (extension == ".tex" || extension == ".latex") {
68 return ExportFormat::LaTeX;
69 } else {
70 // Default to CSV if unknown
71 return ExportFormat::CSV;
72 }
73}
