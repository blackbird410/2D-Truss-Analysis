18VectorXd DirectSolver::solve(const MatrixXd& A, const VectorXd& b) const {
19 // Validate dimensions
20 if (A.rows() != A.cols()) {
21 throw std::runtime_error("DirectSolver: Matrix A must be square");
22 }
23 if (A.rows() != b.size()) {
24 throw std::runtime_error("DirectSolver: Matrix dimensions incompatible with vector");
25 }
26
27 // Check if matrix is symmetric within a relative tolerance.
28 // 1e-10 is chosen to handle floating-point assembly errors in stiffness
29 // matrices while still detecting genuinely non-symmetric matrices.
30 bool isSymmetric = A.isApprox(A.transpose(), 1e-10);
31
32 if (isSymmetric) {
33 // Use LDLT for symmetric positive definite matrices (more efficient)
34 Eigen::LDLT<MatrixXd> ldlt(A);
35
36 // Check if decomposition was successful
37 if (ldlt.info() != Eigen::Success) {
38 throw std::runtime_error("DirectSolver: LDLT decomposition failed - matrix may be "
39 "singular or not positive definite");
40 }
41
42 // Solve the system
43 VectorXd x = ldlt.solve(b);
44 return x;
45 }
46 // Use FullPivLU for general non-symmetric matrices
47 // This is more robust for ill-conditioned and rank-deficient matrices
48 Eigen::FullPivLU<MatrixXd> lu(A);
49
50 // Check if matrix is singular
51 if (!lu.isInvertible()) {
52 throw std::runtime_error("DirectSolver: Matrix is singular or rank-deficient");
53 }
54
55 // Solve the system
56 VectorXd x = lu.solve(b);
57 return x;
58}
