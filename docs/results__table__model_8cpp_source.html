55QVariant ResultsTableModel::data(const QModelIndex& index, int role) const {
56 if (!index.isValid())
57 return {};
58
59 const int row = index.row();
60 const int col = index.column();
61
62 if (row < 0 || row >= static_cast<int>(m_rows.size()))
63 return {};
64 if (col < 0 || col >= kColumnCount)
65 return {};
66
67 if (role == Qt::DisplayRole) {
68 const auto& pair = m_rows[static_cast<size_t>(row)];
69 return (col == kColKey) ? pair.first : pair.second;
70 }
71
72 if (role == Qt::TextAlignmentRole) {

85void ResultsTableModel::refresh(const IAnalysisResultsView& view) {
86 beginResetModel();
87 m_rows.clear();
88
89 auto add = [this](const char* key, const QString& val) {
90 m_rows.push_back({QString::fromUtf8(key), val});
91 };
92
93 // ---- Solver status ----
94 add("Converged",
95 view.hasConverged() ? QStringLiteral("Yes \u2713") : QStringLiteral("No \u2717"));
96 add("Iterations", QString::number(view.getIterations()));
97 add("Residual Norm", QString::number(view.getResidualNorm(), 'e', 4));
98 add("Condition Number", QString::number(view.getConditionNumber(), 'g', 6));
99
100 // ---- Peak response ----
101 add("Max Displacement",
102 QStringLiteral("%1 mm").arg(view.getMaxDisplacement() * 1000.0, 0, 'f', 4));
103 add("Max Stress", QStringLiteral("%1 MPa").arg(view.getMaxStress() / 1.0e6, 0, 'f', 3));
104 add("Total Strain Energy", QStringLiteral("%1 J").arg(view.getTotalStrain(), 0, 'g', 6));
105
106 // ---- Degrees of freedom ----
107 add("Total DOFs", QString::number(static_cast<long long>(view.getTotalDofs())));
108 add("Free DOFs", QString::number(static_cast<long long>(view.getFreeDofs())));
109 add("Constrained DOFs", QString::number(static_cast<long long>(view.getConstrainedDofs())));
110
111 endResetModel();
112}
