25bool XMLExporter::exportResults(const ITrussView& truss,
26 const IAnalysisResultsView& results,
27 const std::filesystem::path& filePath,
28 const ExportOptions& options) {
29 std::ofstream file(filePath);
30 if (!file.is_open()) {
31 m_lastError = "Cannot open file for writing: " + filePath.string();
32 return false;
33 }
34
35 try {
36 // XML declaration
37 file << "<?xml version=\"1.0\" encoding=\"UTF-8\"?>\n";
38 file << "<TrussAnalysisResults>\n";
39
40 // Project metadata
41 file << " <Project>\n";
42 file << " <Name>" << truss::utils::string::escapeXml(truss.getName()) << "</Name>\n";
43 file << " <ExportTime>" << truss::utils::string::formatTimestamp() << "</ExportTime>\n";
44 file << " <Version>3.0.0</Version>\n";
45 file << " </Project>\n";
46
47 // Geometry section (conditional)
48 if (options.includeGeometry) {
49 writeGeometrySection(file, truss, options);
50 }
51
52 // Properties section (conditional)
53 if (options.includeProperties) {
54 writePropertiesSection(file, truss, options);
55 }
56
57 // Loads section (conditional)
58 if (options.includeLoads) {
59 writeLoadsSection(file, truss, options);
60 }
61
62 // Displacements section (conditional)
63 if (options.includeDisplacements && results.getDisplacements().size() > 0) {
64 writeDisplacementsSection(file, results, options);
65 }
66
67 // Member forces section (conditional)
68 if (options.includeMemberForces && !results.getMemberForces().empty()) {
69 writeMemberForcesSection(file, results, options);
70 }
71
72 // Reactions section (conditional)
73 if (options.includeReactions && !results.getReactions().empty()) {
74 writeReactionsSection(file, results, options);
75 }
76
77 // Metadata section (conditional)
78 if (options.includeMetadata) {
79 writeMetadataSection(file, results, options);
80 }
81
82 file << "</TrussAnalysisResults>\n";
83
84 file.close();
85 return true;
86
87 } catch (const std::exception& e) {
88 m_lastError = "Export failed: " + std::string(e.what());
89 return false;
90 }
91}

93void XMLExporter::writeGeometrySection(std::ostream& os,
94 const ITrussView& truss,
95 const ExportOptions& options) {
96 os << " <Geometry>\n";
97
98 // Nodes
99 os << " <Nodes>\n";
100 for (const auto& node : truss.getNodeViews()) {
101 os << " <Node id=\"" << node.id << "\">\n";
102 os << " <X>"
103 << truss::utils::string::formatReal(
104 node.x, options.precision, options.useScientificNotation)
105 << "</X>\n";
106 os << " <Y>"
107 << truss::utils::string::formatReal(
108 node.y, options.precision, options.useScientificNotation)
109 << "</Y>\n";
110 os << " <SupportType>" << static_cast<int>(node.support) << "</SupportType>\n";
111 os << " </Node>\n";
112 }
113 os << " </Nodes>\n";
114
115 // Members
116 os << " <Members>\n";
117 for (const auto& member : truss.getMemberViews()) {
118 os << " <Member id=\"" << member.id << "\">\n";
119 os << " <StartNode>" << member.startNodeId << "</StartNode>\n";
120 os << " <EndNode>" << member.endNodeId << "</EndNode>\n";
121 os << " <Length>"
122 << truss::utils::string::formatReal(
123 member.length, options.precision, options.useScientificNotation)
124 << "</Length>\n";
125 os << " </Member>\n";
126 }
127 os << " </Members>\n";
128
129 os << " </Geometry>\n";
130}

219 os << " <Force memberId=\"" << (i + 1) << "\" type=\"" << type << "\">";
220 os << truss::utils::string::formatReal(
