83QVariant MemberTableModel::data(const QModelIndex& index, int role) const {
84 if (!index.isValid())
85 return {};
86
87 const int row = index.row();
88 const int col = index.column();
89
90 if (row < 0 || row >= static_cast<int>(m_rows.size()))
91 return {};
92 if (col < 0 || col >= kColumnCount)
93 return {};
94
95 const MemberView& m = m_rows[static_cast<size_t>(row)];
96
97 // ------------------------------------------------------------------
98 // TextAlignmentRole
99 // ------------------------------------------------------------------
100 if (role == Qt::TextAlignmentRole) {
101 if (col == kColId || col == kColMaterial || col == kColState)
102 return static_cast<int>(Qt::AlignCenter);
103 return static_cast<int>(Qt::AlignRight | Qt::AlignVCenter);
104 }
105
106 // ------------------------------------------------------------------
107 // ForegroundRole — tension/compression/yield colour coding
108 // Colors mirror the canvas member colours so the table and canvas tell
109 // the same visual story. Utilisation ratio column keeps the green→amber→red
110 // interpolation; force and stress columns use the state-based palette.
111 // ------------------------------------------------------------------
112 if (role == Qt::ForegroundRole) {
113 if (!m_hasResults)
114 return {};
115 if (col == kColRatio)
116 return ratioColor(m.utilizationRatio);
117 // Force and stress: colour by mechanical state
118 if (col == kColForce || col == kColStress) {
119 if (m.yielded)
120 return QColor(0xFF, 0x17, 0x44); // red — yielded
121 if (m.inTension)
122 return QColor(0x4F, 0xC3, 0xF7); // cyan — tension
123 if (m.axialForce < -1e-10)
124 return QColor(0xFF, 0x70, 0x43); // orange — compression
125 }
126 return {};
127 }
128
129 // ------------------------------------------------------------------
130 // BackgroundRole — semi-transparent state tint on the State column
131 // A light tint provides a quick at-a-glance row colour cue that reads
132 // correctly in both dark and light themes at alpha ≈40.
133 // ------------------------------------------------------------------
134 if (role == Qt::BackgroundRole && m_hasResults) {
135 if (col == kColState) {
136 if (m.yielded)
137 return QBrush(QColor(0xFF, 0x17, 0x44, 45));
138 if (m.inTension)
139 return QBrush(QColor(0x4F, 0xC3, 0xF7, 45));
140 if (m.axialForce < -1e-10)
141 return QBrush(QColor(0xFF, 0x70, 0x43, 45));
142 }
143 return {};
144 }
145
146 // ------------------------------------------------------------------
147 // ToolTipRole — state description (col 11 only)
148 // ------------------------------------------------------------------
149 if (role == Qt::ToolTipRole) {
150 if (col == kColState && m_hasResults)
151 return stateString(m);
152 if (col == kColE)
153 return QStringLiteral("Young's Modulus");
154 if (col == kColA)
155 return QStringLiteral("Cross-sectional area");
156 return {};
157 }
158
159 // ------------------------------------------------------------------
160 // DisplayRole
161 // ------------------------------------------------------------------
162 if (role != Qt::DisplayRole)
163 return {};
164
165 // Results columns: show "—" when no analysis has been run
166 const bool isResultsCol = (col >= kColForce && col <= kColState);
167 if (isResultsCol && !m_hasResults)
168 return QStringLiteral("—");
169
170 switch (col) {
171 case kColId:
172 return QString::number(m.id);
173
174 case kColStart:
175 return QString::number(m.startNodeId);
176
177 case kColEnd:
178 return QString::number(m.endNodeId);
179
180 case kColMaterial:
181 return QString::fromStdString(m.label);
182
183 case kColE:
184 // Pa → GPa
185 return QString::number(m.youngModulus / 1.0e9, 'f', 1);
186
187 case kColA:
188 // m² → cm² (1 m² = 1×10⁴ cm²)
189 return QString::number(m.area * 1.0e4, 'f', 4);
190
191 case kColLength:
192 return QString::number(m.length, 'f', 4);
193
194 case kColAngle: {
195 const double deg = m.angle * (180.0 / std::numbers::pi);
196 return QString::number(deg, 'f', 2);
197 }
198
199 case kColForce:
200 // N → kN
201 return QString::number(m.axialForce / 1000.0, 'f', 3);
202
203 case kColStress:
204 // Pa → MPa
205 return QString::number(m.axialStress / 1.0e6, 'f', 3);
206
207 case kColRatio:
208 return QString::number(m.utilizationRatio, 'f', 3);
209
210 case kColState:
211 return stateString(m);
212
213 default:
214 return {};
215 }

246QColor MemberTableModel::ratioColor(double ratio) const {
247 ratio = std::clamp(ratio, 0.0, 1.0);
248
249 auto lerp = [](int a, int b, double t) -> int {
250 return static_cast<int>(std::round(a + (b - a) * t));
251 };
252
253 QColor c1, c2;
254 double t = 0.0;
255
256 if (ratio <= 0.75) {
257 c1 = kColorGreen;
258 c2 = kColorAmber;
259 t = ratio / 0.75;
260 } else {
261 c1 = kColorAmber;
262 c2 = kColorRed;
263 t = (ratio - 0.75) / 0.25;
264 }
265
266 return QColor(lerp(c1.red(), c2.red(), t),
267 lerp(c1.green(), c2.green(), t),
268 lerp(c1.blue(), c2.blue(), t));
269}
