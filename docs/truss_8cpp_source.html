93bool Truss::updateNode(NodeId nodeId, const Point2D& newPosition) {
94 auto node = getNode(nodeId);
95 if (!node) {
96 return false;
97 }
98 // Update position in-place. The node's ID, support type, and applied load
99 // are untouched. Members holding a shared_ptr to this Node see the new
100 // coordinates immediately (length/angle are computed on demand from the
101 // node positions), so no member-level bookkeeping is needed.
102 node->setPosition(newPosition);
103 return true;
104}

106bool Truss::updateMember(MemberId memberId,
107 const MaterialProperties& material,
108 const SectionProperties& section) {
109 auto member = getMember(memberId);
110 if (!member) {
111 return false;
112 }
113 // Replace material and section in-place. The member's ID and node
114 // connectivity (shared_ptr<Node> startNode / endNode) are preserved;
115 // only the structural properties that feed into the stiffness matrix
116 // are changed. The caller must re-run analysis to obtain updated results.
117 member->setMaterial(material);
118 member->setSection(section);
119 return true;
120}

180std::vector<NodePtr> Truss::getNodesInRegion(const Point2D& bottomLeft,
181 const Point2D& topRight) const {
182 std::vector<NodePtr> result;
183 std::copy_if(m_nodes.begin(),
184 m_nodes.end(),
185 std::back_inserter(result),
186 [&bottomLeft, &topRight](const auto& node) {
187 Real x = node->getX();
188 Real y = node->getY();

199size_t Truss::getFreeDofs() const {
200 size_t freeDofs = 0;
201 for (const auto& node : m_nodes) {
202 SupportType support = node->getSupportType();
203
204 // Count free X DOF
205 if (support == SupportType::Free || support == SupportType::RollerX) {
206 freeDofs++;
207 }
208
209 // Count free Y DOF
210 if (support == SupportType::Free || support == SupportType::RollerY) {
211 freeDofs++;
212 }
213 }
214 return freeDofs;

345Point2D Truss::getCentroid() const {
346 if (m_nodes.empty())
347 return Point2D(0, 0);
348
349 Real sumX{0.0};
350 Real sumY{0.0};

372Truss::TrussStatistics Truss::getStatistics() const {
373 TrussStatistics stats;
374 stats.totalNodes = getNodeCount();
375 stats.totalMembers = getMemberCount();
376 stats.freeDofs = getFreeDofs();
377 stats.constrainedDofs = getConstrainedDofs();
378 stats.totalWeight = getTotalWeight();
379 stats.boundingBoxMin = getBoundingBoxMin();
380 stats.boundingBoxMax = getBoundingBoxMax();
381
382 // Count applied forces
383 for (const auto& node : m_nodes) {
384 if (node->hasAppliedForce()) {
385 stats.appliedForces++;
386 }
387 }
388
389 // Calculate total length
390 for (const auto& member : m_members) {
391 stats.totalLength += member->getLength();
392 }
393
394 return stats;
395}
