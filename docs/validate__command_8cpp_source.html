138bool ValidateCommand::validateInputFile() const {
139 namespace fs = std::filesystem;
140
141 try {
142 if (!fs::exists(m_inputFile)) {
143 return false;
144 }
145
146 if (!fs::is_regular_file(m_inputFile)) {
147 return false;
148 }
149
150 // Check read permissions
151 auto perms = fs::status(m_inputFile).permissions();
152 return (perms & fs::perms::owner_read) != fs::perms::none;
153
154 } catch (const std::filesystem::filesystem_error&) {
155 return false;
156 }
157}
