157void NotificationRail::addNotification(NotificationSeverity severity, const QString& message) {
158 // Evict the oldest item if already at the limit.
159 while (m_items.size() >= kMaxItems) {
160 removeItem(m_items.first().data());
161 }
162
163 auto* item = new NotificationItem(severity, message, this);
164 m_layout->addWidget(item);
165 m_items.append(QPointer<QWidget>(item));
166
167 // Close button → immediate removal.
168 connect(item, &NotificationItem::closeRequested, this, [this, item]() { removeItem(item); });
169
170 // Auto-dismiss timer for non-Error severity.
171 if (severity != NotificationSeverity::Error) {
172 QTimer::singleShot(kAutoDismissMs, this, [this, item]() { removeItem(item); });
173 }
174
175 // Slide-in animation: maximumHeight 0 → kItemHeight over kAnimDurationMs.
176 auto* anim = new QPropertyAnimation(item, "maximumHeight", item);
177 anim->setDuration(kAnimDurationMs);
178 anim->setStartValue(0);
179 anim->setEndValue(kItemHeight);
180 anim->setEasingCurve(QEasingCurve::OutCubic);
181 anim->start(QAbstractAnimation::DeleteWhenStopped);
182
183 // Ensure the rail is visible once we have content.
184 setVisible(true);
185}

187void NotificationRail::removeItem(QWidget* item) {
188 if (!item) {
189 return;
190 }
191
192 // Remove from tracking list.
193 m_items.removeIf([item](const QPointer<QWidget>& p) { return p.data() == item; });
194
195 item->deleteLater();
196 Q_EMIT notificationDismissed();
197
198 // Hide the rail when no more items remain.
199 if (m_items.isEmpty()) {
200 setVisible(false);
201 }
202}
