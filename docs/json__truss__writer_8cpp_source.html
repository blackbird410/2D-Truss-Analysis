22bool JsonTrussWriter::write(const core::interfaces::TrussDTO& trussData,
23 const std::filesystem::path& filepath,
24 const FileIOOptions& options) {
25 // Check if file exists and overwrite not allowed
26 if (std::filesystem::exists(filepath) && !options.overwriteExisting) {
27 throw FileWriteException(filepath.string() + " (file exists, overwrite not allowed)");
28 }
29
30 // Note: validateOnWrite flag exists but is not enforced here.
31 // Infrastructure layer works with DTOs (data), not Domain objects.
32 // Validation should occur at Application layer using TrussValidator
33 // BEFORE calling write(). Tests set validateOnWrite=false to skip
34 // application-level validation when testing pure I/O serialization.
35
36 // Create JSON structure
37 json j;
38
39 if (options.includeMetadata) {
40 j["metadata"] = createMetadata(trussData, options);
41 }
42
43 j["nodes"] = createNodes(trussData);
44 j["members"] = createMembers(trussData);
45 j["loads"] = createLoads(trussData);
46
47 // Write to file
48 try {
49 std::ofstream file(filepath);
50 if (!file.is_open()) {
51 throw FileWriteException(filepath.string());
52 }
53
54 if (options.prettyPrint) {
55 file << j.dump(options.indentSize);
56 } else {
57 file << j.dump();
58 }
59
60 file.close();
61 return true;
62 } catch (const std::exception& e) {
63 throw FileWriteException(filepath.string() + " (" + e.what() + ")");
64 }
65}

67json JsonTrussWriter::createMetadata(const core::interfaces::TrussDTO& trussData,
68 [[maybe_unused]] const FileIOOptions& options) {
69 json metadata;
70 metadata["name"] = trussData.name;
71 metadata["version"] = "3.0.0";
72 metadata["format"] = "truss-json-v1";
73
74 // Add timestamp
75 auto now = std::chrono::system_clock::now();
76 auto time = std::chrono::system_clock::to_time_t(now);
77 std::ostringstream oss;
78 oss << std::put_time(std::localtime(&time), "%Y-%m-%d %H:%M:%S");
79 metadata["timestamp"] = oss.str();
80
81 return metadata;
82}

100json JsonTrussWriter::createMembers(const core::interfaces::TrussDTO& trussData) {
101 json membersArray = json::array();
102
103 for (const auto& member : trussData.members) {
104 json memberObj;
105 memberObj["id"] = member.id;
106 memberObj["startNode"] = member.startNodeId;
107 memberObj["endNode"] = member.endNodeId;
108
109 // Material properties (flattened from DTO)
110 json materialObj;
111 materialObj["youngsModulus"] = member.youngModulus;
112 materialObj["density"] = member.density;
113 materialObj["yieldStrength"] = member.yieldStrength;
114 memberObj["material"] = materialObj;
115
116 // Section properties (flattened from DTO)
117 json sectionObj;
118 sectionObj["area"] = member.area;
119 memberObj["section"] = sectionObj;
120
121 membersArray.push_back(memberObj);
122 }
123
124 return membersArray;
125}

127json JsonTrussWriter::createLoads(const core::interfaces::TrussDTO& trussData) {
128 json loadsArray = json::array();
129
130 for (const auto& node : trussData.nodes) {
131 // Only write non-zero forces
132 if (std::abs(node.fx) > 1e-10 || std::abs(node.fy) > 1e-10) {
133 json loadObj;
134 loadObj["nodeId"] = node.id;
135 loadObj["fx"] = node.fx;
136 loadObj["fy"] = node.fy;
137
138 loadsArray.push_back(loadObj);
139 }
140 }
141
142 return loadsArray;
143}
