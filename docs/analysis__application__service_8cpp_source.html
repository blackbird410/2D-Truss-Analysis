19Result<ResultsHandle>
20AnalysisApplicationService::analyze(const core::Truss& truss,
21 const core::analysis::AnalysisOptions& options) {
22 try {
23 // Phase 1: Validate truss structure
24 auto validationResult = core::validation::TrussValidator::validate(truss);
25 if (!validationResult.isValid()) {
26 std::string errorMsg = "Truss validation failed:\n";
27 for (const auto& message : validationResult.getErrorMessages()) {
28 errorMsg += " - " + message + "\n";
29 }
30 return Result<ResultsHandle>::Failure(errorMsg);
31 }
32
33 // Phase 2: Create orchestrator with solver and validator
34 // Note: AnalysisOrchestrator takes ownership of solver and validator
35 // Select solver based on options.useDirectSolver
36 std::unique_ptr<core::analysis::ILinearSolver> solver;
37 if (options.useDirectSolver) {
38 solver = std::make_unique<core::analysis::DirectSolver>();
39 } else {
40 solver = std::make_unique<core::analysis::IterativeSolver>();
41 }
42 auto validator = std::make_unique<core::validation::TrussValidator>();
43 core::analysis::AnalysisOrchestrator orchestrator(
44 std::move(solver), std::move(validator), options);
45
46 // Phase 3: Run analysis on a local mutable copy
47 core::Truss trussCopy = truss;
48 auto results = orchestrator.analyze(trussCopy);
49
50 // Phase 4: Store results and generate handle
51 auto handle = generateHandle();
52 m_results[handle] = std::make_shared<core::analysis::AnalysisResults>(std::move(results));
53
54 return Result<ResultsHandle>::Success(handle);
55
56 } catch (const std::exception& e) {
57 return Result<ResultsHandle>::Failure(std::string("Analysis failed: ") + e.what());
58 }
59}

81Result<bool>
82AnalysisApplicationService::exportResults(ResultsHandle handle,
83 truss::ExportFormat format,
84 const std::filesystem::path& filepath,
85 const core::Truss& truss,
86 const infrastructure::export_::ExportOptions& options) {
87 try {
88 // Validate handle
89 if (!isValidResultsHandle(handle)) {
90 return Result<bool>::Failure("Invalid results handle: " + std::to_string(handle));
91 }
92
93 // Get results
94 const auto& results = getResults(handle);
95
96 // Create exporter for specified format
97 auto exporter = infrastructure::export_::ExporterFactory::create(format);
98
99 // Export results - check return value for file I/O errors
100 bool exportSuccess = exporter->exportResults(truss, results, filepath, options);
101 if (!exportSuccess) {
102 std::string errorMsg = exporter->getLastError();
103 if (errorMsg.empty()) {
104 errorMsg = "Export failed for unknown reason";
105 }
106 return Result<bool>::Failure(errorMsg);
107 }
108
109 return Result<bool>::Success(true);
110
111 } catch (const std::exception& e) {
112 return Result<bool>::Failure(std::string("Export failed: ") + e.what());
113 }
114}
