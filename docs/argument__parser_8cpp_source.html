15ParsedArgs ArgumentParser::parse(
16 int argc,
17 const char* const
18 argv[]) { // NOLINT(cppcoreguidelines-avoid-c-arrays,modernize-avoid-c-arrays)
19 ParsedArgs result;
20
21 // Skip program name (argv[0])
22 for (int i = 1; i < argc; ++i) {
23 std::string arg = argv[i];
24
25 // Check for help flag
26 if (arg == "-h" || arg == "--help") {
27 result.showHelp = true;
28 result.commandName = "help";
29 continue;
30 }
31
32 // Check for verbose flag
33 if (arg == "-v" || arg == "--verbose") {
34 result.verbose = true;
35 result.options["verbose"] = "true";
36 continue;
37 }
38
39 // Check for other options
40 if (isOption(arg)) {
41 std::string optionName = extractOptionName(arg);
42 std::string optionValue = extractOptionValue(arg);
43
44 // If value not in same argument, check next argument
45 if (optionValue.empty() && i + 1 < argc && !isOption(argv[i + 1])) {
46 optionValue = argv[++i];
47 }
48
49 result.options[optionName] = optionValue.empty() ? "true" : optionValue;
50 } else {
51 // Positional argument - first one is command name
52 if (result.commandName.empty()) {
53 result.commandName = arg;
54 } else {
55 result.positionalArgs.push_back(arg);
56 }
57 }
58 }
59
60 // Default command if none specified
61 if (result.commandName.empty() && !result.showHelp) {
62 result.commandName = "help"; // Show help by default
63 }
64
65 return result;
66}

82std::string ArgumentParser::extractOptionName(const std::string& arg) {
83 if (isLongOption(arg)) {
84 // Extract from --name or --name=value
85 size_t start = 2;
86 size_t end = arg.find('=');
87 return arg.substr(start, end == std::string::npos ? end : end - start);
88 }
89 if (isShortOption(arg)) {
90 // Extract from -n or -n=value
91 size_t start = 1;
92 size_t end = arg.find('=');
93 return arg.substr(start, end == std::string::npos ? end : end - start);
94 }
95 return "";
96}

98std::string ArgumentParser::extractOptionValue(const std::string& arg) {
99 size_t pos = arg.find('=');
100 if (pos != std::string::npos && pos + 1 < arg.size()) {
101 return arg.substr(pos + 1);
102 }
103 return "";
104}

106std::optional<std::string> ArgumentParser::getOption(const ParsedArgs& args,
107 const std::string& longForm,
108 const std::string& shortForm) {
109 // Try long form first
110 auto it = args.options.find(longForm);
111 if (it != args.options.end()) {
112 return it->second;
113 }
114
115 // Try short form if long form not found
116 it = args.options.find(shortForm);
117 if (it != args.options.end()) {
118 return it->second;
119 }
120
121 // Not found
122 return std::nullopt;
123}
