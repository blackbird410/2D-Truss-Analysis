70std::shared_ptr<core::Truss> TrussBuilder::build() {
71 // Validate minimum requirements
72 if (m_nodes.empty()) {
73 throw std::runtime_error("Cannot build truss: no nodes defined. Add at least 2 nodes.");
74 }
75 if (m_nodes.size() < 2) {
76 throw std::runtime_error("Cannot build truss: need at least 2 nodes, got " +
77 std::to_string(m_nodes.size()));
78 }
79 if (m_members.empty()) {
80 throw std::runtime_error("Cannot build truss: no members defined. Add at least 1 member.");
81 }
82
83 // Create the truss object
84 auto truss = std::make_shared<core::Truss>(m_name);
85
86 // Add all nodes (NodeId assignment happens automatically in Truss::addNode)
87 for (const auto& nodeData : m_nodes) {
88 truss->addNode(nodeData.x, nodeData.y, nodeData.support);
89 }
90
91 // Add all members
92 for (const auto& memberData : m_members) {
93 // Convert application DTOs to domain types (following TrussApplicationService pattern)
94 core::MaterialProperties material{
95 memberData.material.youngsModulusPa, // E
96 7850.0, // density (default for steel)
97 memberData.material.youngsModulusPa * 0.00125, // yield strength (estimate)
98 memberData.material.youngsModulusPa * 0.002, // ultimate strength (estimate)
99 memberData.material.name // name
100 };
101
102 core::SectionProperties section{
103 memberData.section.areaM2, // area
104 memberData.section.areaM2 * memberData.section.areaM2 /
105 12.0, // moment of inertia (estimate)
106 memberData.section.areaM2, // shear area
107 memberData.section.profile // designation
108 };
109
110 truss->addMember(memberData.startNodeId, memberData.endNodeId, material, section);
111 }
112
113 // Apply all forces
114 for (const auto& forceData : m_forces) {
115 truss->applyForce(forceData.nodeId, forceData.force);
116 }
117
118 // Note: We don't call reset() here to allow inspection of what was built
119 // Users can explicitly call reset() if they want to reuse the builder
120 return truss;
121}

138void TrussBuilder::validateNodeReference(core::NodeId nodeId, const std::string& context) const {
139 if (!isValidNodeId(nodeId)) {
140 std::ostringstream oss;
141 oss << "Invalid node reference in " << context << ": node ID " << nodeId;
142 if (m_nodes.empty()) {
143 oss << " (no nodes defined yet)";
144 } else {
145 oss << " (valid range: 1-" << (m_nextNodeId - 1) << ")";
146 }
147 throw std::invalid_argument(oss.str());
148 }
149}
