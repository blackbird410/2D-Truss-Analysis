60void ConsoleLogger::log(LogLevel level, const std::string& message) const {
61 std::lock_guard<std::mutex> lock(m_mutex);
62
63 // Check level filter (duplicates isLevelEnabled logic to avoid deadlock,
64 // since isLevelEnabled() also acquires m_mutex)
65 if (static_cast<int>(level) < static_cast<int>(m_minLevel)) {
66 return;
67 }
68
69 // Build log line
70 std::ostringstream oss;
71 oss << "[" << getCurrentTimestamp() << "] " << getColorCode(level) << getLevelString(level)
72 << getResetCode() << " " << message;
73
74 std::string logLine = oss.str();
75
76 // Output to appropriate stream
77 std::FILE* stream = (level >= LogLevel::Error) ? stderr : stdout;
78 std::fwrite(logLine.c_str(), 1, logLine.size(), stream);
79 std::fwrite("\n", 1, 1, stream);
80 std::fflush(stream);
81}

83std::string ConsoleLogger::getCurrentTimestamp() {
84 auto now = std::chrono::system_clock::now();
85 auto time = std::chrono::system_clock::to_time_t(now);
86
87 // Use thread-safe time conversion
88 std::tm timeInfo;
89 localtime_r(&time, &timeInfo);
90
91 std::ostringstream oss;
92 oss << std::put_time(&timeInfo, "%Y-%m-%d %H:%M:%S");
93 return oss.str();
