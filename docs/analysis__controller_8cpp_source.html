79void AnalysisController::onAnalyzeRequested(const core::analysis::AnalysisOptions& opts) {
80 if (m_trussHandle == 0) {
81 emit analysisFailed(QStringLiteral("No active truss to analyze."));
82 return;
83 }
84
85 if (m_thread && m_thread->isRunning()) {
86 // Analysis already in progress — ignore duplicate request.
87 return;
88 }
89
90 // ---- Step 1: copy the Truss on the main thread ----
91 truss::core::Truss trussCopy = m_facade.getTrussMutable(m_trussHandle);
92
93 // ---- Step 2: announce start ----
94 emit analysisStarted();
95
96 // ---- Step 3: set up worker + thread ----
97 m_thread = new QThread{this};
98
99 // IAnalysisService is inherited by ITrussAnalysisFacade
100 auto* service = static_cast<truss::application::IAnalysisService*>(&m_facade);
101
102 auto* worker = new AnalysisWorker{std::move(trussCopy), opts, service};
103 worker->moveToThread(m_thread);
104
105 // Worker lifetime: deleted when thread finishes
106 connect(m_thread, &QThread::finished, worker, &QObject::deleteLater);
107
108 // Results routing (QueuedConnection — worker thread → main thread)
109 connect(worker,
110 &AnalysisWorker::finished,
111 this,
112 &AnalysisController::onWorkerFinished,
113 Qt::QueuedConnection);
114 connect(worker,
115 &AnalysisWorker::failed,
116 this,
117 &AnalysisController::onWorkerFailed,
118 Qt::QueuedConnection);
119
120 // Start execution
121 connect(m_thread, &QThread::started, worker, &AnalysisWorker::execute);
122 m_thread->start();
123}
